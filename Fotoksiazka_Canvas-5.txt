// Minimalny parser EXIF (DateTimeOriginal) – bez zewnętrznych bibliotek.
async function readExifDate(file) {
  try {
    if (!(file.type === 'image/jpeg' || /\.jpe?g$/i.test(file.name))) return null;
    const v = new DataView(await file.slice(0, 262144).arrayBuffer());
    if (v.getUint16(0) !== 0xFFD8) return null;
    let off = 2;
    while (off + 4 < v.byteLength) {
      const marker = v.getUint16(off); off += 2;
      if (marker === 0xFFDA || (marker & 0xFF00) !== 0xFF00) break;
      const len = v.getUint16(off);
      if (marker === 0xFFE1 && v.getUint32(off + 2) === 0x45786966) return parseTiffDate(v, off + 8);
      off += len;
    }
  } catch (e) { /* brak EXIF */ }
  return null;
}
function parseTiffDate(v, tiff) {
  const little = v.getUint16(tiff) === 0x4949;
  const u16 = o => v.getUint16(o, little), u32 = o => v.getUint32(o, little);
  if (u16(tiff + 2) !== 42) return null;
  const readStr = (o, n) => { let s = ''; for (let i = 0; i < n - 1; i++) { const c = v.getUint8(o + i); if (!c) break; s += String.fromCharCode(c); } return s; };
  const readIFD = (ifdOff, wanted) => {
    const out = {}, n = u16(tiff + ifdOff);
    for (let i = 0; i < n; i++) {
      const e = tiff + ifdOff + 2 + i * 12, tag = u16(e), type = u16(e + 2), cnt = u32(e + 4);
      if (!wanted.includes(tag)) continue;
      if (type === 2) out[tag] = readStr(cnt > 4 ? tiff + u32(e + 8) : e + 8, cnt);
      else if (type === 4) out[tag] = u32(e + 8);
      else if (type === 3) out[tag] = u16(e + 8);
    }
    return out;
  };
  const ifd0 = readIFD(u32(tiff + 4), [0x0132, 0x8769]);
  const exif = ifd0[0x8769] ? readIFD(ifd0[0x8769], [0x9003, 0x9004]) : {};
  return exif[0x9003] || exif[0x9004] || ifd0[0x0132] || null;
}

function exifOrientation(v, off, end) {
  try {
    if (v.getUint32(off) !== 0x45786966) return 0;                    // 'Exif'
    const t = off + 6, le = v.getUint16(t) === 0x4949;
    if (v.getUint16(t + 2, le) !== 42) return 0;
    const ifd = t + v.getUint32(t + 4, le), n = v.getUint16(ifd, le);
    for (let i = 0; i < n && ifd + 2 + i * 12 + 12 <= end; i++) {
      const e = ifd + 2 + i * 12;
      if (v.getUint16(e, le) === 0x0112) return v.getUint16(e + 8, le);
    }
  } catch (e) { /* nietypowy EXIF */ }
  return 0;
}
// Wymiary JPEG z nagłówka (bez dekodowania), już PO uwzględnieniu obrotu – null, gdy to nie JPEG.
async function jpegSize(blob) {
  try {
    const v = new DataView(await blob.slice(0, 131072).arrayBuffer());
    if (v.getUint16(0) !== 0xFFD8) return null;
    let o = 2, ori = 0;
    while (o + 9 < v.byteLength) {
      if (v.getUint8(o) !== 0xFF) { o++; continue; }
      const m = v.getUint8(o + 1);
      if (m === 0xD8 || m === 0x01 || (m >= 0xD0 && m <= 0xD7)) { o += 2; continue; }
      const len = v.getUint16(o + 2);
      if (m === 0xE1 && !ori) ori = exifOrientation(v, o + 4, Math.min(v.byteLength, o + 2 + len));
      if (m >= 0xC0 && m <= 0xCF && m !== 0xC4 && m !== 0xC8 && m !== 0xCC) {
        const h = v.getUint16(o + 5), w = v.getUint16(o + 7);
        return ori >= 5 && ori <= 8 ? { w: h, h: w, ori } : { w, h, ori };    // 5-8: obrót o 90°, boki zamienione
      }
      o += 2 + len;
    }
  } catch (e) { /* nie JPEG */ }
  return null;
}

        dateStr = `${m[1]}.${m[2]}.${m[3]}`; timeStr = `${m[4]}:${m[5]}`;
        ts = new Date(+m[1], +m[2] - 1, +m[3], +m[4], +m[5], +(m[6] || 0)).getTime();
      } else {
        const d = new Date(ts);
        dateStr = `${d.getFullYear()}.${String(d.getMonth() + 1).padStart(2, '0')}.${String(d.getDate()).padStart(2, '0')}`;
        timeStr = null;                                   // bez EXIF nie znamy godziny – nie wymyślamy jej
        exifError = true;
      }
      added.push({
        id: ++uid, file: f, name: f.name, thumbUrl, w: w0, h: h0, aspect: w0 / h0,
        orientation: w0 > h0 ? 'landscape' : 'portrait', dateStr, timeStr, timestamp: ts, exifError,
        frame: null, stamp: null, plan: null, slot: null
      });
    } catch (err) {
      bad++; log(`Pominięto ${f.name}: ${err.message}`, 'warn');
    }
    await yieldUi();
  }
  // Nowe zdjęcia wchodzą chronologicznie, ale nie ruszają Twojej ręcznej kolejności.
  added.sort((a, b) => a.timestamp - b.timestamp);
  for (const p of added) {
    let idx = photos.length;
    while (idx > 0 && photos[idx - 1].timestamp > p.timestamp) idx--;
    photos.splice(idx, 0, p);
  }
  if (bad) toast(`Nie udało się wczytać ${bad} plik(ów) – szczegóły w dzienniku.`);
  if (files.length > take.length) toast(`Wczytano tyle, ile mieści limit ${MAX_PHOTOS} zdjęć.`);
}

// Limit czasu odpowiedzi: zawieszone zapytanie nie może zablokować generowania (potem jest ponawiane).
const REQ_TIMEOUT_MS = { image: 300000, text: 150000 };
async function rawPost(model, body, signal) {
  const wait = cooldownUntil - Date.now();
  if (wait > 0) await sleepAbortable(wait, signal);
  // W Canvas adres MUSI być bez klucza: przechwytywanie działa tylko dla zapytań bez parametru „key”.
  // Adres z PUSTYM parametrem „key” — dokładnie tak, jak w działającej wersji aplikacji.
  // Canvas rozpoznaje po nim zapytanie i wykonuje je na koncie użytkownika; bez tego parametru wraca 401.
  const url = `${API_BASE}/models/${model}:generateContent?key=${API_KEY}`;
  const isImg = !!IMAGE_MODELS[model], ms = isImg ? REQ_TIMEOUT_MS.image : REQ_TIMEOUT_MS.text;
  noteRequest(isImg ? 'image' : 'text');
  const ctl = new AbortController(); let timedOut = false;
  const onAbort = () => ctl.abort();
  if (signal) { if (signal.aborted) throw abortErr(); signal.addEventListener('abort', onAbort, { once: true }); }
  const timer = setTimeout(() => { timedOut = true; ctl.abort(); }, ms);
  try {
    const r = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body), signal: ctl.signal });
    const text = await r.text();
    let json = null; try { json = JSON.parse(text); } catch (_) { /* nie-JSON */ }
    const out = { ok: r.ok, status: r.status, json, text };
    noteResult(out);
    return out;
  } catch (e) {
    if (timedOut && !(signal && signal.aborted)) { const te = new Error(`przekroczono limit czasu odpowiedzi API (${Math.round(ms / 1000)} s)`); te.timeout = true; throw te; }
    throw e;
  } finally { clearTimeout(timer); if (signal) signal.removeEventListener('abort', onAbort); }
}
const errMsg = res => (res.json && res.json.error && res.json.error.message) || res.text.slice(0, 240) || ('HTTP ' + res.status);

// Do północy czasu pacyficznego (wtedy odnawiają się dzienne limity API) – w minutach.
function minutesToPacificMidnight() {
  try {
    const parts = new Intl.DateTimeFormat('en-US', { timeZone: 'America/Los_Angeles', hour12: false, hour: '2-digit', minute: '2-digit' }).formatToParts(new Date());
    const h = (+parts.find(p => p.type === 'hour').value) % 24, m = +parts.find(p => p.type === 'minute').value;
    return 24 * 60 - (h * 60 + m);
  } catch (e) { return 360; }
}
// Wyciąga z odpowiedzi 429 dokładną nazwę i wartość przekroczonego limitu (QuotaFailure) oraz czas oczekiwania.
function parseQuota(res, model) {
  const err = (res.json && res.json.error) || {}, msg = String(err.message || res.text || '');
  const q = { model: model || null, id: '', metric: '', value: null, retryMs: retryDelayMs(res), msg, at: Date.now(), status: res.status, violations: [] };
  for (const d of err.details || []) {
    if (/QuotaFailure/.test(d['@type'] || '')) for (const v of d.violations || []) {
      q.violations.push({ id: v.quotaId || '', metric: v.quotaMetric || '', value: v.quotaValue != null ? String(v.quotaValue) : null, model: (v.quotaDimensions && v.quotaDimensions.model) || null });
    }
  }
  const pick = q.violations.find(v => v.value === '0') || q.violations[0];
  if (pick) { q.id = pick.id; q.metric = pick.metric; q.value = pick.value; q.model = pick.model || q.model; }
  let m;
  if (q.value == null && (m = /limit:\s*(\d+)/i.exec(msg))) q.value = m[1];
  if (!q.metric && (m = /metric:\s*([\w./\-]+)/i.exec(msg))) q.metric = m[1];
  if (!q.id && (m = /Quota ID:\s*([\w\-]+)/i.exec(msg))) q.id = m[1];
  if (!q.model && (m = /model:\s*([\w.\-]+)/i.exec(msg))) q.model = m[1];
  const blob = `${q.id} ${q.metric} ${msg}`;
  q.zero = q.value === '0' || q.violations.some(v => v.value === '0') || /limit:\s*0\b/i.test(msg);
  q.daily = /PerDay|per_day|per day/i.test(blob);
  q.minute = /PerMinute|per_minute|per minute/i.test(blob);
  q.free = /FreeTier|free_tier/i.test(blob);
  q.spend = /spend/i.test(blob);
  q.short = [q.model, q.id || q.metric || null, q.value != null ? `limit ${q.value}` : null].filter(Boolean).join(' · ') || 'API nie podało szczegółów';
  return q;
}

  if (q.zero) return { level: 'bad', title: 'Limit 0 – brak przydziału, a nie wyczerpanie',

  if (q.daily) return { level: 'bad', title: 'Wyczerpany limit dzienny',
    lines: [`Konto wyczerpało dzienny limit dla tego modelu${model}.`,

  return { level: 'warn', title: 'Limit chwilowy',
    lines: ['Za dużo zapytań naraz. Aplikacja sama zwalnia tempo i ponawia — zwykle wystarczy poczekać.',
      'Możesz też zmniejszyć „Równoległe zapytania” w kroku 04.'], short: q.short };

function quotaFatal(chain) {
  const isImg = chain.some(m => IMAGE_MODELS[m]), q = USAGE.lastQuota;
  const zero = q && q.zero;
  return fatal(isImg
    ? (zero ? 'Twoje konto Gemini nie ma przydziału na modele obrazu. Wybierz lżejszy model w kroku 04 albo spróbuj później.'
      : 'Modele obrazu odmówiły rysowania (limit). Aplikacja czekała i ponawiała — spróbuj ponownie za kilka minut.')
    : 'Modele tekstowe odmówiły odpowiedzi (limit). Opowieści dni można wyłączyć w kroku 03.');
}

// Dowolny obraz z API -> JPEG (mniej pamięci niż PNG).
async function callGemini({ chain, bodiesFor, extract, signal, tag }) {
  let lastErr = new Error('Brak odpowiedzi z API');
  let blocked = 0;
  const noQuotaLeft = () => chain.every(m => isDead(m) && (deadReason[m] === 'zero' || deadReason[m] === 'daily'));
  if (noQuotaLeft()) throw quotaFatal(chain);                 // nie ma sensu pytać ponownie – od razu jasny komunikat
  const alive = chain.filter(m => !isDead(m));
  for (const model of (alive.length ? alive : chain)) {
    const bodies = bodiesFor(model);
    let dead = false;
    for (let li = Math.min(levelMemo[model] || 0, bodies.length - 1); li < bodies.length && !dead; li++) {
      let nextLevel = false;
      for (let attempt = 0; attempt < 5 && !nextLevel && !dead; attempt++) {
        if (signal && signal.aborted) throw abortErr();
        let res;
        try { res = await rawPost(model, bodies[li], signal); }
        catch (e) {
          if (e.name === 'AbortError') throw e;
          lastErr = new Error('Błąd sieci: ' + e.message); log(`${tag}: ${lastErr.message}`, 'warn');
          if (e.timeout && attempt >= 1) { dead = true; continue; }          // dwa zawieszone zapytania: próbujemy następnego modelu
          await sleepAbortable(1500 * (attempt + 1), signal); continue;
        }
        if (res.ok) {
          const out = extract(res.json);
          if (out) { levelMemo[model] = li; return { value: out, model }; }
          lastErr = new Error(emptyReason(res.json));
          if (isBlocked(res.json) && ++blocked >= 2) { lastErr.blocked = true; throw lastErr; }
          log(`${tag}: ${lastErr.message} (${model}, próba ${attempt + 1})`, 'warn');
          await sleepAbortable(800 * (attempt + 1), signal); continue;
        }
        const msg = errMsg(res);
        lastErr = new ApiError(`${model}: ${msg}`, res.status);
        if (res.status === 400) {
          if (/API key not valid|API_KEY_INVALID|API key expired/i.test(res.text)) {
            throw fatal('Gemini Canvas nie przekazał zapytania do Twojego konta. Odśwież stronę Canvas i spróbuj ponownie.');
          }
          log(`${tag}: wariant żądania ${li + 1}/${bodies.length} odrzucony (400, ${model}) – to nieszkodliwe, próbuję następnego. Powód: ${msg.slice(0, 200)}`, 'warn');
          nextLevel = true; continue;
        }
        if (res.status === 401 || res.status === 403 || res.status === 404) {
          log(`${tag}: model ${model} niedostępny w tym środowisku (${res.status}) – próbuję następnego. ${msg.slice(0, 140)}`, 'warn');
          markDead(model, res.status === 401 ? 5 : 15); dead = true; continue;
        }
        if (res.status === 429) {
          const q = parseQuota(res, model); USAGE.lastQuota = q;
          if (q.zero || q.daily) {                             // przydział zerowy albo dzienny: czekanie nic nie da – następny model
            deadReason[model] = q.zero ? 'zero' : 'daily';
            markDead(model, q.zero ? 60 : Math.min(minutesToPacificMidnight(), 360));
            log(`${tag}: ${q.zero ? 'brak przydziału (limit: 0)' : 'limit dzienny wyczerpany'} – ${q.short}. Pomijam ten model.`, 'warn');
            dead = true; continue;
          }
          const wait = Math.min(90000, q.retryMs || 4000 * 2 ** attempt);   // limit chwilowy: czekamy, ile każe API, i zwalniamy tempo
          cooldownUntil = Math.max(cooldownUntil, Date.now() + wait);
          if (concCap > 1) { concCap = 1; log('Limit chwilowy: zmniejszam liczbę równoległych zapytań do 1.', 'warn'); }
          log(`${tag}: limit chwilowy (${q.short}) – czekam ${Math.round(wait / 1000)} s`, 'warn');
          await sleepAbortable(wait, signal); continue;
        }
        if (res.status >= 500) {
          log(`${tag}: błąd serwera ${res.status}, ponawiam`, 'warn');
          await sleepAbortable(2000 * 2 ** Math.min(attempt, 3), signal); continue;
        }
        dead = true;
      }
      if (!nextLevel) dead = true;                    // wyczerpano próby dla tego modelu
    }
  }
  if (noQuotaLeft()) throw quotaFatal(chain);
  if (lastErr.status === 401) throw fatal('Żaden model nie odpowiedział — Canvas nie przekazał zapytań do Twojego konta (401). Odśwież stronę Canvas i uruchom aplikację ponownie.');
  throw lastErr;
}

async function genImage({ parts, ratio, size, S, signal, tag }) {
  const chain = S.autoFallback ? [S.imageModel, ...IMAGE_ORDER.filter(m => m !== S.imageModel)] : [S.imageModel];
  const r = await callGemini({ chain, signal, tag, bodiesFor: m => imageBodies(m, parts, ratio, size, S), extract: extractImage });
  return { b64: r.value.data, mime: r.value.mime, model: r.model };
}
async function genText({ parts, signal, tag }) {
  const r = await callGemini({ chain: TEXT_MODELS, signal, tag, bodiesFor: () => [{ contents: [{ role: 'user', parts }] }], extract: extractText });
  return r.value;
}
// Dowolny obraz z API -> JPEG (mniej pamięci niż PNG).

  if (S.category !== 'custom') return S.category;
  const raw = S.customTheme;
  if (!raw) return 'Dowolny kreatywny motyw w stylu Pixar';

  for (const part of raw.split(',').map(p => p.trim()).filter(Boolean)) {

  }
  if (specific) return specific;
  return general.length ? general.join(', ') : raw;

function rrectPath(g, x, y, w, h, r) {
  r = Math.max(0, Math.min(r, w / 2, h / 2));
  g.beginPath(); g.moveTo(x + r, y);
  g.arcTo(x + w, y, x + w, y + h, r); g.arcTo(x + w, y + h, x, y + h, r);
  g.arcTo(x, y + h, x, y, r); g.arcTo(x, y, x + w, y, r); g.closePath();
}
// Geometria „okna” w kadrze: proporcje zdjęcia, zadany udział POWIERZCHNI ramki,
// margines dzielony 62/38 w pionie (więcej u góry – tam wychodzi głowa postaci).
const WIN_MARGIN = 0.022;           // minimalny pas dekoracji przy krawędzi (ułamek krótszego boku)
function templateGeom(ratio, photoAspect, pct, LONG) {
  const R = ratioVal(ratio);
  const W = R >= 1 ? LONG : Math.round(LONG * R), H = R >= 1 ? Math.round(LONG / R) : LONG;
  const m = Math.min(W, H) * WIN_MARGIN;
  let ww = Math.sqrt(clamp(pct, 0.2, 0.97) * W * H * photoAspect), wh = ww / photoAspect;
  const s = Math.min(1, (W - 2 * m) / ww, (H - 2 * m) / wh);               // nie wychodzimy poza margines
  ww *= s; wh *= s;
  return { W, H, x: (W - ww) / 2, y: (H - wh) * 0.62, w: ww, h: wh };
}
// Szablon wysyłany do modelu: biała płachta + magentowe okno. Twoje zdjęcie NIE opuszcza urządzenia.
function makeTemplate(ratio, photoAspect, pct) {
  const t = templateGeom(ratio, photoAspect, pct, 1024);
  const c = canvasOf(t.W, t.H), g = c.getContext('2d');
  g.fillStyle = '#ffffff'; g.fillRect(0, 0, t.W, t.H);
  g.fillStyle = '#ff00ff'; g.fillRect(Math.round(t.x), Math.round(t.y), Math.round(t.w), Math.round(t.h));
  return c.toDataURL('image/png').split(',')[1];
}

// Znajduje magentowe okno w obrazie z AI i buduje maskę (bez obwódki na krawędziach).
function detectWindow(img, photoAspect) {
  const iw = img.width, ih = img.height, sc = Math.min(1, 640 / Math.max(iw, ih));
  const w = Math.max(16, Math.round(iw * sc)), h = Math.max(16, Math.round(ih * sc)), N = w * h;
  const c = canvasOf(w, h), g = c.getContext('2d', { willReadFrequently: true });
  g.drawImage(img, 0, 0, w, h);
  const d = g.getImageData(0, 0, w, h).data;
  const key = new Uint8Array(N), loose = new Uint8Array(N);
  for (let p = 0, i = 0; p < N; p++, i += 4) {
    const r = d[i], gg = d[i + 1], b = d[i + 2];
    if (r > 165 && b > 165 && gg < 120 && r - gg > 85 && b - gg > 85) key[p] = 1;                 // czysta magenta
    if (r - gg > 40 && b - gg > 40 && Math.abs(r - b) < 110) loose[p] = 1;                        // jej cieniowane odcienie
  }
  const lab = new Int32Array(N), st = new Int32Array(N);
  let bestArea = 0, bestId = 0, id = 0;
  for (let s = 0; s < N; s++) {                                                                  // największa spójna plama
    if (!key[s] || lab[s]) continue;
    id++; let sp = 0, area = 0; st[sp++] = s; lab[s] = id;
    while (sp) {
      const p = st[--sp]; area++; const x = p % w;
      if (x > 0 && key[p - 1] && !lab[p - 1]) { lab[p - 1] = id; st[sp++] = p - 1; }
      if (x < w - 1 && key[p + 1] && !lab[p + 1]) { lab[p + 1] = id; st[sp++] = p + 1; }
      if (p >= w && key[p - w] && !lab[p - w]) { lab[p - w] = id; st[sp++] = p - w; }
      if (p < N - w && key[p + w] && !lab[p + w]) { lab[p + w] = id; st[sp++] = p + w; }
    }
    if (area > bestArea) { bestArea = area; bestId = id; }
  }
  if (!bestId || bestArea < N * 0.03) return { ok: false, why: 'nie znaleziono okna magenta' };
  let x0 = w, y0 = h, x1 = -1, y1 = -1;
  for (let p = 0; p < N; p++) if (lab[p] === bestId) { const x = p % w, y = (p / w) | 0; if (x < x0) x0 = x; if (x > x1) x1 = x; if (y < y0) y0 = y; if (y > y1) y1 = y; }
  const bw = x1 - x0 + 1, bh = y1 - y0 + 1, fill = bestArea / (bw * bh);
  if (fill < 0.7) return { ok: false, why: `okno ma nieregularny kształt (${Math.round(fill * 100)}%)` };
  if (Math.abs(Math.log((bw / bh) / photoAspect)) > 0.3) return { ok: false, why: 'okno zmieniło proporcje' };
  const M = new Uint8Array(N);
  for (let y = y0; y <= y1; y++) for (let x = x0; x <= x1; x++) { const p = y * w + x; if (lab[p] === bestId || loose[p]) M[p] = 1; }
  const R = 2, tmp = new Uint8Array(N), out = new Uint8Array(N);                                 // poszerzenie maski o 2 px
  for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) { let v = 0; for (let k = Math.max(0, x - R); k <= Math.min(w - 1, x + R); k++) if (M[y * w + k]) { v = 1; break; } tmp[y * w + x] = v; }
  for (let x = 0; x < w; x++) for (let y = 0; y < h; y++) { let v = 0; for (let k = Math.max(0, y - R); k <= Math.min(h - 1, y + R); k++) if (tmp[k * w + x]) { v = 1; break; } out[y * w + x] = v; }
  const mc = canvasOf(w, h), mg = mc.getContext('2d'), md = mg.createImageData(w, h);
  for (let p = 0; p < N; p++) md.data[p * 4 + 3] = out[p] ? 255 : 0;
  mg.putImageData(md, 0, 0);
  const sx = iw / w, sy = ih / h;
  return { ok: true, mask: mc, dil: R, sx, sy, fill, bbox: { x: x0 * sx, y: y0 * sy, w: bw * sx, h: bh * sy } };
}

// Limity pamięci przy składaniu ramki (dłuższy bok / liczba pikseli). Słabsze urządzenia dostają niższy sufit.
const COMPOSE_MAX_LONG = 8192;
const composeMaxArea = () => (navigator.deviceMemory && navigator.deviceMemory < 6) ? 20e6 : 32e6;
const downSize = sz => ({ '4K': '2K', '2K': '1K' }[sz] || sz);
// Składanie ramek jedna po drugiej: duże płótna nie kumulują się w pamięci nawet przy kilku zapytaniach naraz.

// Mała miniatura (kafelki „ściany”, lista) – żeby przeglądarka nie dekodowała wielomegapikselowych obrazów dla ikonki.
async function canvasThumbBlob(src, max = 360, q = 0.8) {

}

}

// Wstawia ORYGINALNE zdjęcie w okno. Twarze pozostają 1:1, a zdjęcie zachowuje NATYWNĄ rozdzielczość:
// ramkę z AI powiększamy tak, by okno pomieściło wszystkie piksele zdjęcia (w granicach pamięci urządzenia).
async function composeSafe(photo, raw, det) {
  const W = raw.width, H = raw.height, bb = det.bbox;
  const ph = await decodeBitmap(photo.file);
  let out = null, layer = null;
  try {
    const sMin = Math.max(W, H) >= 1800 ? 1 : 2;
    const sNative = Math.max(ph.width / bb.w, ph.height / bb.h);
    const sCap = Math.max(1, Math.min(COMPOSE_MAX_LONG / Math.max(W, H), Math.sqrt(composeMaxArea() / (W * H))));
    const S = Math.min(Math.max(sMin, sNative), sCap), OW = Math.round(W * S), OH = Math.round(H * S);
    out = canvasOf(OW, OH); const o = out.getContext('2d');
    o.imageSmoothingQuality = 'high'; o.drawImage(raw, 0, 0, OW, OH);
    const pad = Math.ceil(det.dil * Math.max(det.sx, det.sy) * S) + 1;
    const x0 = Math.max(0, Math.floor(bb.x * S - pad)), y0 = Math.max(0, Math.floor(bb.y * S - pad));
    const x1 = Math.min(OW, Math.ceil((bb.x + bb.w) * S + pad)), y1 = Math.min(OH, Math.ceil((bb.y + bb.h) * S + pad));
    const bw = x1 - x0, bh = y1 - y0;
    layer = canvasOf(bw, bh); const l = layer.getContext('2d');           // warstwa tylko w obrębie okna (mniej pamięci)
    l.imageSmoothingQuality = 'high';
    l.fillStyle = '#ffffff'; l.fillRect(0, 0, bw, bh);
    const mode = Math.abs(Math.log((ph.width / ph.height) / (bw / bh))) < 0.045 ? 'cover' : 'contain';
    const r = fitInto(ph.width, ph.height, 0, 0, bw, bh, mode);
    l.drawImage(ph, r.x, r.y, r.w, r.h);
    if (det.fill < 0.985) {                                               // okno nieprostokątne: zachowujemy elementy sceny nachodzące na okno
      const mw = det.mask.width, mh = det.mask.height;
      l.globalCompositeOperation = 'destination-in';
      l.drawImage(det.mask, x0 / OW * mw, y0 / OH * mh, bw / OW * mw, bh / OH * mh, 0, 0, bw, bh);
    }
    o.drawImage(layer, x0, y0);
    const blob = await toBlob(out, 'image/jpeg', 0.93), thumb = await canvasThumbBlob(out);
    return { blob, thumb, w: OW, h: OH };
  } finally {
    if (ph.close) ph.close();
    if (out) out.width = out.height = 1;
    if (layer) layer.width = layer.height = 1;
  }
}

function dropBmp(key) {                                   // usuwa wszystkie warianty rozmiaru danego klucza
  for (const k of Array.from(bmpCache.keys())) if (k === key || k.startsWith(key + '@')) { const b = bmpCache.get(k); if (b && b.close) b.close(); bmpCache.delete(k); }
}
async function cachedBitmap(key, blob, maxW) {
  const ck = key + '@' + (maxW || 0);
  if (bmpCache.has(ck)) { const v = bmpCache.get(ck); bmpCache.delete(ck); bmpCache.set(ck, v); return v; }
  const bmp = await decodeScaled(blob, maxW);                // podgląd nie potrzebuje pełnej rozdzielczości
  bmpCache.set(ck, bmp);
  while (bmpCache.size > 16) { const k0 = bmpCache.keys().next().value, b = bmpCache.get(k0); if (b && b.close) b.close(); bmpCache.delete(k0); }
  return bmp;
}
function setFrame(photo, f) {

  dropBmp('f' + photo.id);
  photo.frame = Object.assign({}, f, { url: URL.createObjectURL(f.blob), thumbUrl: f.thumb ? URL.createObjectURL(f.thumb) : null });
}

/* ===================== RAMKA NA CAŁĄ STRONĘ (wariant opcjonalny) ===================== */
// Cała kartka to jedna kompozycja: dekoracja generowana osobno, zdjęcia wklejane programowo NAD nią.
// Dotychczasowy wariant ramek pozostaje nietknięty — ta ścieżka jest całkowicie równoległa.
const PAGE_TARGET = 0.90;           // cel: zdjęcia zajmują 90% kartki, reszta to grafika
const PAGE_BAND_MAX = 0.075;        // najszerszy pas dekoracji przy krawędzi
const PAGE_BAND_MIN = 0.015;        // najwęższy (~3 mm na A4) – gdy trzeba dociągnąć do celu
const PAGE_FOOT_U = 62;             // pas u dołu na numer strony w jednostkach strony (pastylka 54 + odstęp)
const PAGE_GAP = 0.022;             // odstęp między dwoma zdjęciami
function punchInfo(S, ori) {
  const base = S.pdfFormat.startsWith('a4') ? [210, 297] : [420, 594];
  const mm = ori === 'landscape' ? { w: base[1], h: base[0] } : { w: base[0], h: base[1] };
  const side = S.pdfFormat === 'a4_holes_left' ? 'left' : S.pdfFormat === 'a4_holes_top' ? 'top' : null;
  return { side, mm: 20, page: mm };
}
// Proporcje okien: pojedyncze zdjęcie można położyć na boku. Kadr zostaje nietknięty, zmienia się tylko ułożenie.
const winAspects = (page, rot) => page.photos.map(x => rot ? 1 / x.aspect : x.aspect);
// Okna wpisane w zadany prostokąt roboczy, z zachowaniem proporcji zdjęć.
function windowsIn(page, x0, y0, bw, bh, gap, rot) {
  const A = winAspects(page, rot);
  if (bw <= 0 || bh <= 0) return A.map(() => ({ x: x0, y: y0, w: 1, h: 1 }));
  if (A.length === 1) return [fitInto(A[0], 1, x0, y0, bw, bh, 'contain')];
  if (page.layout === 'col') {                                  // dwa zdjęcia jedno pod drugim
    const hFit = (bh - gap) / 2, w = Math.min(bw, hFit * Math.min(A[0], A[1]));
    const hs = A.map(a => w / a), th = hs[0] + hs[1] + gap, sc = Math.min(1, bh / th);
    const ww = w * sc, h2 = A.map(a => ww / a), tot = h2[0] + h2[1] + gap;
    let y = y0 + (bh - tot) / 2; const x = x0 + (bw - ww) / 2;
    return A.map((a, i) => { const r = { x, y, w: ww, h: h2[i] }; y += h2[i] + gap; return r; });
  }
  const wFit = (bw - gap) / 2, h = Math.min(bh, wFit / Math.max(A[0], A[1]));   // dwa zdjęcia obok siebie
  const ws = A.map(a => h * a), tw = ws[0] + ws[1] + gap, sc = Math.min(1, bw / tw);
  const hh = h * sc, w2 = A.map(a => hh * a), tot = w2[0] + w2[1] + gap;
  let x = x0 + (bw - tot) / 2; const y = y0 + (bh - hh) / 2;
  return A.map((a, i) => { const r = { x, y, w: w2[i], h: hh }; x += w2[i] + gap; return r; });
}
// Okna na zdjęcia w jednostkach strony. Pasy dekoracji zwężają się tylko tam, gdzie naprawdę ograniczają
// wielkość zdjęć — po drugiej stronie zostaje szeroki margines na ozdoby.
function pageWindows(page, S, ori) {
  const P = PAGE[ori], pi = punchInfo(S, ori), upm = P.w / pi.page.w;
  const punch = pi.side ? Math.round(pi.mm * upm) : 0;
  const gap = Math.round(Math.min(P.w, P.h) * PAGE_GAP);
  const build = (bx, by, rot) => {
    const px = Math.round(P.w * bx), py = Math.round(P.h * by);
    const x0 = px + (pi.side === 'left' ? punch : 0), y0 = py + (pi.side === 'top' ? punch : 0);
    return windowsIn(page, x0, y0, P.w - px - x0, P.h - py - PAGE_FOOT_U - y0, gap, rot);
  };
  const area = w => w.reduce((a, r) => a + r.w * r.h, 0) / (P.w * P.h);
  // Kładziemy kadr na boku TYLKO wtedy, gdy realnie zyskuje powierzchnię. Kwadrat nic nie zyskuje, więc zostaje prosto.
  const rot = page.photos.length === 1
    && area(build(PAGE_BAND_MIN, PAGE_BAND_MIN, true)) > area(build(PAGE_BAND_MIN, PAGE_BAND_MIN, false)) * 1.02;
  page.winRot = rot;
  const steps = [];
  for (let b = PAGE_BAND_MAX; b >= PAGE_BAND_MIN - 1e-9; b -= 0.002) steps.push(b);
  const celny = steps.find(b => area(build(b, b, rot)) >= PAGE_TARGET);
  if (celny) return build(celny, celny, rot);                      // cel osiągnięty – zostawiamy jak najszerszą dekorację
  const aMin = area(build(PAGE_BAND_MIN, PAGE_BAND_MIN, rot));     // proporcje nie pozwalają na 90%: zdjęcia tak duże, jak się da,
  const bx = steps.find(b => area(build(b, PAGE_BAND_MIN, rot)) >= aMin * 0.999) || PAGE_BAND_MIN;  // a wolne miejsce zostaje na grafikę
  const aBx = area(build(bx, PAGE_BAND_MIN, rot));
  const by = steps.find(b => area(build(bx, b, rot)) >= aBx * 0.999) || PAGE_BAND_MIN;
  return build(bx, by, rot);
}
// Szablon całej kartki: biała płachta, magentowe okna i jasny pas zaznaczający margines pod dziurkacz.
function makePageTemplate(wins, S, ori) {
  const P = PAGE[ori], ratio = pickRatio(P.w / P.h), rv = ratioVal(ratio), LONG = 1024;
  const W = rv >= 1 ? LONG : Math.round(LONG * rv), H = rv >= 1 ? Math.round(LONG / rv) : LONG;
  const sx = P.w / W, sy = P.h / H;
  const c = canvasOf(W, H), g = c.getContext('2d');
  g.fillStyle = '#ffffff'; g.fillRect(0, 0, W, H);
  const pi = punchInfo(S, ori), upm = P.w / pi.page.w;
  if (pi.side) {
    g.fillStyle = '#ededed';
    if (pi.side === 'left') g.fillRect(0, 0, Math.round(pi.mm * upm / sx), H);
    else g.fillRect(0, 0, W, Math.round(pi.mm * upm / sy));
  }
  g.fillStyle = '#ff00ff';
  wins.forEach(r => g.fillRect(Math.round(r.x / sx), Math.round(r.y / sy), Math.round(r.w / sx), Math.round(r.h / sy)));
  try { return { b64: c.toDataURL('image/png').split(',')[1], ratio }; } finally { c.width = c.height = 1; }
}
// Jak detectWindow, ale szuka kilku okien naraz i zwraca ich położenie w ułamkach obrazu.
function detectWindows(img, want, aspects) {
  const iw = img.width, ih = img.height, sc = Math.min(1, 720 / Math.max(iw, ih));
  const w = Math.max(16, Math.round(iw * sc)), h = Math.max(16, Math.round(ih * sc)), N = w * h;
  const c = canvasOf(w, h), g = c.getContext('2d', { willReadFrequently: true });
  g.drawImage(img, 0, 0, w, h);
  const d = g.getImageData(0, 0, w, h).data;
  c.width = c.height = 1;
  const key = new Uint8Array(N);
  for (let p = 0, i = 0; p < N; p++, i += 4) {
    const r = d[i], gg = d[i + 1], b = d[i + 2];
    if (r > 165 && b > 165 && gg < 120 && r - gg > 85 && b - gg > 85) key[p] = 1;
  }
  const lab = new Int32Array(N), st = new Int32Array(N), blobs = [];
  let id = 0;
  for (let s0 = 0; s0 < N; s0++) {
    if (!key[s0] || lab[s0]) continue;
    id++; let sp = 0, area = 0, x0 = w, y0 = h, x1 = -1, y1 = -1;
    st[sp++] = s0; lab[s0] = id;
    while (sp) {
      const q = st[--sp]; area++;
      const x = q % w, y = (q / w) | 0;
      if (x < x0) x0 = x; if (x > x1) x1 = x; if (y < y0) y0 = y; if (y > y1) y1 = y;
      if (x > 0 && key[q - 1] && !lab[q - 1]) { lab[q - 1] = id; st[sp++] = q - 1; }
      if (x < w - 1 && key[q + 1] && !lab[q + 1]) { lab[q + 1] = id; st[sp++] = q + 1; }
      if (q >= w && key[q - w] && !lab[q - w]) { lab[q - w] = id; st[sp++] = q - w; }
      if (q < N - w && key[q + w] && !lab[q + w]) { lab[q + w] = id; st[sp++] = q + w; }
    }
    const bw = x1 - x0 + 1, bh = y1 - y0 + 1;
    if (area >= N * 0.02 && area / (bw * bh) >= 0.72) blobs.push({ area, x: x0, y: y0, w: bw, h: bh });
  }
  if (blobs.length < want) return { ok: false, why: `znaleziono ${blobs.length} z ${want} okien` };
  blobs.sort((a, b) => b.area - a.area);
  const picked = blobs.slice(0, want);
  const spreadX = Math.max(...picked.map(b => b.x)) - Math.min(...picked.map(b => b.x));
  const spreadY = Math.max(...picked.map(b => b.y)) - Math.min(...picked.map(b => b.y));
  picked.sort((a, b) => spreadX >= spreadY ? a.x - b.x : a.y - b.y);        // kolejność zgodna z układem
  for (let i = 0; i < want; i++) {
    const got = picked[i].w / picked[i].h;
    if (Math.abs(Math.log(got / aspects[i])) > 0.33) return { ok: false, why: `okno ${i + 1} zmieniło proporcje` };
  }
  return { ok: true, wins: picked.map(b => ({ x: b.x / w, y: b.y / h, w: b.w / w, h: b.h / h })) };
}

// Czy zapamiętane okna odpowiadają obecnemu układowi? Chroni przed dekoracją z wcześniejszej wersji aplikacji.
function winsMatch(wins, page) {
  if (!Array.isArray(wins) || wins.length !== page.photos.length) return false;
  const P = PAGE[layoutState.orientation], k = P.w / P.h;        // okna zapisane są w ułamkach obrazu – przeliczamy na proporcje strony
  const chce = winAspects(page, page.winRot);
  return wins.every((r, i) => r && r.h > 0 && Math.abs(Math.log((r.w / r.h * k) / chce[i])) < 0.33);
}
async function pageFrameTask(page, S, signal, tile) {
  tile.set('run');
  const aspects = winAspects(page, page.winRot);
  let done = null, lastErr = null;
  for (let a = 1; a <= 3 && !done; a++) {
    try {
      if (a === 1 && S.useCache !== false) {
        const hit = await cacheGet(cacheKeyPage(page, S));
        if (hit && hit.blob && !winsMatch(hit.wins, page)) log(`Strona ${page.tag}: zapamiętana dekoracja ma inny układ okien – generuję nową`, 'warn');
        else if (hit && hit.blob) {
          setDeco(page, { blob: hit.blob, thumb: hit.thumb, w: hit.w, h: hit.h, kind: 'ai', model: hit.model, wins: hit.wins });
          tile.set('ok', { url: page.deco.thumbUrl || page.deco.url, tag: 'z pamięci' });
          log(`Strona ${page.tag}: dekoracja z pamięci urządzenia (bez zapytania)`);
          return;
        }
      }
      const Sa = a === 3 ? Object.assign({}, S, { imageSize: downSize(S.imageSize) }) : S;
      const gen = await genPageDeco(page, Sa, signal);
      const raw = await createImageBitmap(b64ToBlob(gen.b64, gen.mime));
      try {
        const det = detectWindows(raw, page.photos.length, aspects);
        if (!det.ok) { lastErr = new Error(det.why); log(`Strona ${page.tag}: ${det.why} (próba ${a}/3)`, 'warn'); continue; }
        done = await oneAtATime(() => packPageDeco(raw, det.wins));
        done.wins = det.wins; done.model = gen.model;
      } finally { if (raw.close) raw.close(); }
    } catch (e) {
      if (e.name === 'AbortError') throw e;
      lastErr = e; log(`Strona ${page.tag}: ${e.message}`, 'error');
      if (e.fatal || e.blocked) break;
    }
  }
  if (done) {
    setDeco(page, Object.assign({ kind: 'ai' }, done));
    cachePut(cacheKeyPage(page, S), { blob: done.blob, thumb: done.thumb, w: done.w, h: done.h, model: done.model, wins: done.wins });
    tile.set('ok', { url: page.deco.thumbUrl || page.deco.url, tag: (IMAGE_MODELS[done.model] || {}).short || done.model, model: done.model });
  } else {
    await makeLocalPageDeco(page, S);
    tile.set('err', { url: page.deco.thumbUrl || page.deco.url, tag: 'lokalna', msg: lastErr ? lastErr.message : 'błąd' });
    if (lastErr && lastErr.fatal) throw lastErr;
  }
}

async function frameTask(photo, S, signal, tile) {
  tile.set('run');
  let done = null, lastErr = null;
  const tries = S.frameMode === 'safe' ? 3 : 2;
  for (let a = 1; a <= tries && !done; a++) {
    try {
      if (a === 1 && S.useCache !== false) {                    // gotowa ramka z wcześniejszej sesji: zero zapytań
        const hit = await cacheGet(cacheKey(photo, S));
        if (hit && hit.blob) {
          setFrame(photo, { blob: hit.blob, thumb: hit.thumb, w: hit.w, h: hit.h, kind: 'ai', model: hit.model });
          tile.set('ok', { url: photo.frame.thumbUrl || photo.frame.url, tag: 'z pamięci' });
          log(`Ramka ${photo.n}: wzięta z pamięci urządzenia (bez zapytania)`);
          return;
        }
      }
      const Sa = (a > 1 && a === tries) ? Object.assign({}, S, { imageSize: downSize(S.imageSize) }) : S;   // ostatnia próba: niższa rozdzielczość
      if (Sa !== S && Sa.imageSize !== S.imageSize) log(`Ramka ${photo.n}: ostatnia próba w ${Sa.imageSize}`, 'warn');
      const gen = await genFrameImage(photo, Sa, signal);
      const raw = await createImageBitmap(b64ToBlob(gen.b64, gen.mime));
      try {
        if (S.frameMode === 'safe') {
          const det = detectWindow(raw, photo.aspect);
          if (!det.ok) { lastErr = new Error(det.why); log(`Ramka ${photo.n}: ${det.why} (próba ${a}/${tries})`, 'warn'); continue; }
          const chce = templateGeom(photo.ratio, photo.aspect, Sa.windowPct, 1024);   // czy model nie zmniejszył okna?
          const oczek = chce.w * chce.h / (chce.W * chce.H), mamy = det.bbox.w * det.bbox.h / (raw.width * raw.height);
          if (mamy < oczek * 0.8) log(`Ramka ${photo.n}: model zmniejszył okno — zdjęcie zajmie ${Math.round(mamy * 100)}% kadru zamiast ${Math.round(oczek * 100)}%. Możesz powtórzyć tę ramkę przyciskiem ⟳.`, 'warn');
          done = await oneAtATime(() => composeSafe(photo, raw, det));
        } else done = await oneAtATime(() => packFull(raw));
        done.model = gen.model;
      } finally { if (raw.close) raw.close(); }
    } catch (e) {
      if (e.name === 'AbortError') throw e;
      lastErr = e; log(`Ramka ${photo.n}: ${e.message}`, 'error');
      if (e.fatal || e.blocked) break;
    }
  }
  if (done) {
    setFrame(photo, { blob: done.blob, thumb: done.thumb, w: done.w, h: done.h, kind: 'ai', model: done.model });
    cachePut(cacheKey(photo, S), { blob: done.blob, thumb: done.thumb, w: done.w, h: done.h, model: done.model });   // zapis w tle
    tile.set('ok', { url: photo.frame.thumbUrl || photo.frame.url, tag: (IMAGE_MODELS[done.model] || {}).short || done.model, model: done.model });
  } else {
    await makeLocalFrame(photo, S);
    tile.set('err', { url: photo.frame.thumbUrl || photo.frame.url, tag: 'lokalna', msg: lastErr ? lastErr.message : 'błąd' });
    if (lastErr && lastErr.fatal) throw lastErr;
  }
}

const STORY_FALLBACK = 'Nie udało się wygenerować opowieści tego dnia. Użyj przycisku „Ponów opowieść dnia” pod tą stroną albo spróbuj za chwilę.';

function sampleEvenly(arr, n) {
  if (arr.length <= n) return arr.slice();
  const out = [];
  for (let i = 0; i < n; i++) out.push(arr[Math.round(i * (arr.length - 1) / (n - 1))]);
  return out;
}
// Do opowieści wysyłamy najwyżej 24 zdjęcia z dnia w małym rozmiarze (limit rozmiaru żądania API).
async function genStory(day, S, signal) {
  const sample = sampleEvenly(day.photos, 24);
  const parts = [{ text: storyPrompt(day, S, sample) }];
  for (const p of sample) parts.push({ inlineData: { mimeType: 'image/jpeg', data: await fileToJpegB64(p.file, 640, 0.75) } });
  return genText({ parts, signal, tag: `Opowieść dnia ${day.index}` });
}
// Dzieli opowieść na strony po zdaniach; nic nie ginie (nadmiar trafia na ostatnią stronę).
function chunkStory(text, maxPages) {
  const sentences = text.match(/[^.!?]+[.!?]+["”»)]*\s*/g) || [text];
  const used = sentences.join('').length;
  if (used < text.length) sentences.push(text.slice(used));
  const maxChars = Math.max(550, Math.ceil(text.length / maxPages) + 80);
  const chunks = []; let cur = '';
  for (const s of sentences) {
    if (cur.length + s.length > maxChars && cur.trim()) { chunks.push(cur.trim()); cur = s; } else cur += s;
  }
  if (cur.trim()) chunks.push(cur.trim());
  while (chunks.length > maxPages) { const last = chunks.pop(); chunks[chunks.length - 1] += '\n' + last; }
  return chunks.length ? chunks : [text.trim()];
}

  day.covers = texts.map(t => ({ text: t, blob: null, url: null, model: null }));
}

// Miejsce zdjęcia na stronie (jednostki bazowe 1190×1684 / 1684×1190, jak w oryginale).
// mw/mh = wymiary modułu przed obrotem; rotate = zdjęcie o innej orientacji niż strona obracamy o 90°.
function slotGeom(pageOri, layout, idx, photo) {
  const P = PAGE[pageOri], portrait = pageOri === 'portrait';
  if (layout === 'row') {
    return portrait ? { x: 35 + idx * 570, y: 32, w: 550, h: 1620, rotate: false, mw: 550, mh: 1620 }
      : { x: 32 + idx * 820, y: 35, w: 800, h: 1120, rotate: false, mw: 800, mh: 1120 };
  }
  if (layout === 'col') {
    return portrait ? { x: 35, y: 32 + idx * 820, w: 1120, h: 800, rotate: false, mw: 1120, mh: 800 }
      : { x: 32, y: 35 + idx * 585, w: 1620, h: 550, rotate: false, mw: 1620, mh: 550 };
  }
  const vw = portrait ? 1120 : 1620, vh = portrait ? 1620 : 1120;
  const rotate = photo.orientation !== pageOri;
  return { x: (P.w - vw) / 2, y: (P.h - vh) / 2, w: vw, h: vh, rotate, mw: rotate ? vh : vw, mh: rotate ? vw : vh };
}

    if (p.frame && p.frame.url) URL.revokeObjectURL(p.frame.url);

  days.forEach(d => (d.matrix || []).forEach(pg => {

    dropBmp('d' + pg.id); pg.deco = null;
  }));
  photos.forEach(p => dropBmp('p' + p.id));
  days.forEach(d => (d.covers || []).forEach(c => { if (c.url) { URL.revokeObjectURL(c.url); dropBmp('c' + c.url); } if (c.thumbUrl) URL.revokeObjectURL(c.thumbUrl); }));

}

// Dzieli zdjęcia na dni, układa strony i losuje postać/pozę dla każdego zdjęcia (bez powtórek).
function planLayout(S) {
  resetResults();
  layoutState = { orientation: S.orientation, layoutMode: S.layoutMode, frameStyle: S.frameStyle, punch: punchInfo(S, S.orientation) };
  const map = new Map();
  photos.forEach(p => { if (!map.has(p.dateStr)) map.set(p.dateStr, []); map.get(p.dateStr).push(p); });
  const subjects = new Deck(() => subjectPool(S)), poses = new Deck(() => [...ALL_POSES]);
  days = []; let di = 1, n = 0;
  for (const [key, list] of map) {
    const day = { index: di, key, photos: list, theme: themeForDay(S, di), matrix: [], covers: [], chunks: [], storyText: '', storyFailed: false };
    let i = 0;
    while (i < list.length) {
      const a = list[i], b = list[i + 1];
      let layout = 'single';
      if (b && S.layoutMode === 'classic' && a.orientation === 'portrait' && b.orientation === 'portrait') layout = 'row';
      else if (b && S.layoutMode === 'smart') {
        if (S.orientation === 'portrait' && a.orientation === 'landscape' && b.orientation === 'landscape') layout = 'col';
        else if (S.orientation === 'landscape' && a.orientation === 'portrait' && b.orientation === 'portrait') layout = 'row';
      }
      const arr = layout === 'single' ? [a] : [a, b];
      const pg = { type: 'matrix', day, photos: arr, layout, style: S.frameStyle === 'page' ? 'page' : 'classic' };
      day.matrix.push(pg);
      arr.forEach((p, idx) => { p.slot = slotGeom(S.orientation, layout, idx, p); p.ratio = pickRatio(p.slot.mw / p.slot.mh); });
      if (pg.style === 'page') {                                  // kompozycja obejmująca całą kartkę
        pg.id = `${di}-${day.matrix.length}`;
        pg.tag = `${di}.${day.matrix.length}`;
        pg.wins = pageWindows(pg, S, S.orientation);
      }
      i += arr.length;
    }
    list.forEach((p, k) => {
      p.n = ++n; p.dayIdx = di; p.idxInDay = k;
      const subject = S.animal === 'random' ? subjects.draw()
        : S.animal === 'internet' ? 'INTERNET: Wyszukaj obiekt pasujący do: ' + day.theme : S.animal;
      p.plan = { theme: day.theme, subject, pose: S.pose === 'random' ? poses.draw() : S.pose };
    });
    day.matrix.forEach(pg => { if (pg.style === 'page') pg.plan = pg.photos[0].plan; });
    days.push(day); di++;
  }
}

}

/* ---------- rysowanie (Canvas 2D; wszystko w jednostkach bazowych, skala k) ---------- */
// Cień w canvasie nie skaluje się z transformacją, więc mnożymy go przez k ręcznie.
function shadowOn(g, k, blur, dy, alpha) { g.shadowColor = `rgba(0,0,0,${alpha})`; g.shadowBlur = blur * k; g.shadowOffsetX = 0; g.shadowOffsetY = dy * k; }
function shadowOff(g) { g.shadowColor = 'transparent'; g.shadowBlur = 0; g.shadowOffsetY = 0; }

// Zawijanie tekstu; null = odstęp między akapitami. Nie łamie po niełamliwej spacji.
function wrapLines(g, text, maxW) {
  const lines = [], paras = text.split(/\n+/);
  paras.forEach((para, pi) => {
    let cur = '';
    for (const word of para.split(' ')) {
      const test = cur ? cur + ' ' + word : word;
      if (cur && g.measureText(test).width > maxW) { lines.push(cur); cur = word; } else cur = test;
    }
    lines.push(cur);
    if (pi < paras.length - 1) lines.push(null);
  });
  return lines;
}
// Zmniejsza czcionkę tak, aby tekst zawsze mieścił się w ramce (oryginał ucinał końcówki).
function fitText(g, text, maxW, maxH, startPx, minPx, lh) {
  for (let px = startPx; px >= minPx; px--) {
    g.font = `600 ${px}px ${FONT}`;
    const lines = wrapLines(g, text, maxW);
    if (lines.reduce((s, l) => s + (l === null ? 0.45 : 1), 0) * px * lh <= maxH) return { px, lines };
  }
  g.font = `600 ${minPx}px ${FONT}`;
  return { px: minPx, lines: wrapLines(g, text, maxW) };
}
function drawTextCard(g, k, text, x, y, w, h, startPx, minPx) {
  g.save();
  shadowOn(g, k, 30, 10, 0.07); rrectPath(g, x, y, w, h, 20); g.fillStyle = '#ffffff'; g.fill(); shadowOff(g);
  const pad = 40, lh = 1.8, fit = fitText(g, text, w - 2 * pad, h - 2 * pad, startPx, minPx, lh);
  g.font = `600 ${fit.px}px ${FONT}`; g.fillStyle = '#1f2937'; g.textAlign = 'left'; g.textBaseline = 'top';
  let ty = y + pad;
  for (const line of fit.lines) { if (line === null) { ty += fit.px * lh * 0.45; continue; } g.fillText(line, x + pad, ty); ty += fit.px * lh; }
  g.restore();
}
async function drawIllustration(g, k, cv, x, y, w, h, getBmp) {

  g.lineWidth = 5; g.strokeStyle = '#e5e7eb'; rrectPath(g, x + 2.5, y + 2.5, w - 5, h - 5, 18); g.stroke();
  rrectPath(g, x + 5, y + 5, w - 10, h - 10, 16); g.clip();
  const bmp = cv.blob ? await getBmp('c' + cv.url, cv.blob, Math.ceil(w * k)) : null;
  if (bmp) { const r = fitInto(bmp.width, bmp.height, x + 5, y + 5, w - 10, h - 10, 'contain'); g.drawImage(bmp, r.x, r.y, r.w, r.h); }
  else { g.fillStyle = '#a4b0be'; g.font = `700 24px ${FONT}`; g.textAlign = 'center'; g.textBaseline = 'middle'; g.fillText('Brak ilustracji', x + w / 2, y + h / 2); }
  g.restore();
}

// Rysowanie kompozycji całostronicowej: dekoracja pod spodem, oryginalne zdjęcia nad nią.
// Przy druku z dziurkami parzyste strony dostają dekorację odbitą w poziomie (albo w pionie),
// żeby spokojny margines wypadł po właściwej stronie — zdjęcia i teksty nigdy nie są odbijane.
async function drawPageFrame(g, page, k, ori, getBmp) {
  const P = PAGE[ori], deco = page.deco;
  const side = (layoutState.punch || {}).side;
  const flipX = side === 'left' && page.no % 2 === 0, flipY = side === 'top' && page.no % 2 === 0;
  const wins = (deco && deco.wins && deco.wins.length === page.photos.length)
    ? deco.wins.map(r => ({ x: r.x * P.w, y: r.y * P.h, w: r.w * P.w, h: r.h * P.h }))
    : page.wins;
  const rects = wins.map(r => flipX ? { x: P.w - r.x - r.w, y: r.y, w: r.w, h: r.h }
    : flipY ? { x: r.x, y: P.h - r.y - r.h, w: r.w, h: r.h } : r);
  if (rects.length > 1) {                                     // odbicie zamienia okna miejscami wzdłuż jednej osi
    const dx = Math.abs(wins[0].x - wins[1].x), dy = Math.abs(wins[0].y - wins[1].y);
    const poziomo = dx >= dy;                                 // czy okna stoją obok siebie, czy jedno pod drugim
    if ((flipX && poziomo) || (flipY && !poziomo)) rects.reverse();   // kolejność czytania zostaje bez zmian
  }
  if (deco && deco.blob) {
    const bmp = await getBmp('d' + page.id, deco.blob, Math.ceil(P.w * k));
    g.save();
    if (flipX) { g.translate(P.w, 0); g.scale(-1, 1); }
    if (flipY) { g.translate(0, P.h); g.scale(1, -1); }
    g.drawImage(bmp, 0, 0, P.w, P.h);
    g.restore();
  } else { g.fillStyle = '#ffffff'; g.fillRect(0, 0, P.w, P.h); }
  for (let i = 0; i < page.photos.length; i++) {
    const photo = page.photos[i], box = rects[i];
    g.save();                                               // cień pod kadrem – zdjęcie czytelnie na pierwszym planie
    shadowOn(g, k, 26, 10, 0.32);
    g.fillStyle = '#ffffff'; g.fillRect(box.x - 1, box.y - 1, box.w + 2, box.h + 2);
    g.restore();
    const bmp = await getBmp('p' + photo.id, photo.file, Math.ceil(Math.max(box.w, box.h) * k));
    g.save();
    if (page.winRot) { g.translate(box.x + box.w / 2, box.y + box.h / 2); g.rotate(-Math.PI / 2); }   // kadr położony na boku

  }
}

async function drawMatrix(g, page, k, ori, getBmp) {
  for (const photo of page.photos) {
    const sl = photo.slot;
    g.save();
    g.translate(sl.x + sl.w / 2, sl.y + sl.h / 2);
    if (sl.rotate) g.rotate(-Math.PI / 2);
    let r = { x: -sl.mw / 2, y: -sl.mh / 2, w: sl.mw, h: sl.mh };
    const bmp = photo.frame && photo.frame.blob ? await getBmp('f' + photo.id, photo.frame.blob, Math.ceil(sl.mw * k)) : null;
    if (bmp) { r = fitInto(bmp.width, bmp.height, r.x, r.y, r.w, r.h, 'contain'); g.drawImage(bmp, r.x, r.y, r.w, r.h); }
    else { g.fillStyle = '#f3f4f6'; g.fillRect(r.x, r.y, r.w, r.h); g.fillStyle = '#a4b0be'; g.font = `800 34px ${FONT}`; g.textAlign = 'center'; g.textBaseline = 'middle'; g.fillText('Brak ramki', 0, 0); }
    if (photo.stamp && photo.stamp.blob) await drawStamp(g, k, photo, r, getBmp);
    g.restore();
  }
}
function drawPill(g, k, P, text, compact) {
  g.save();
  g.font = `800 ${compact ? 19 : 24}px ${FONT}`; g.textAlign = 'center'; g.textBaseline = 'middle';
  const w = g.measureText(text).width + (compact ? 34 : 50), h = compact ? 40 : 54;
  const x = (P.w - w) / 2, y = P.h - (compact ? 14 : 30) - h;
  shadowOn(g, k, 20, 10, 0.3); rrectPath(g, x, y, w, h, 20); g.fillStyle = '#8b5cf6'; g.fill(); shadowOff(g);
  g.lineWidth = 3; g.strokeStyle = '#ffffff'; g.stroke();
  g.fillStyle = '#ffffff'; g.fillText(text, P.w / 2, y + h / 2 + 1);
  g.restore();
}

// strip = { y0, h } (w pikselach rastra): rysuje tylko wycinek strony. Cała reszta układu jest liczona identycznie,
// więc sąsiednie paski łączą się bez szwu, a pamięć zależy od wysokości paska, nie od rozmiaru strony.
async function renderPage(page, canvas, pxW, getBmp, strip, bleedFrac) {
  await ensureFonts();
  const ori = layoutState.orientation, P = PAGE[ori];
  const bf = bleedFrac || 0, k = pxW * (1 - 2 * bf) / P.w, off = pxW * bf;   // bf: ułamek szerokości zajęty przez spad
  const fullH = Math.round(P.h * k + 2 * off), y0 = strip ? strip.y0 : 0, h = strip ? strip.h : fullH;
  canvas.width = Math.round(pxW); canvas.height = h;
  const g = canvas.getContext('2d');
  if (!g) throw new Error('przeglądarka odmówiła utworzenia płótna');
  g.fillStyle = '#ffffff'; g.fillRect(0, 0, canvas.width, canvas.height);
  g.setTransform(k, 0, 0, k, off, off - y0); g.imageSmoothingEnabled = true; g.imageSmoothingQuality = 'high';
  if (page.type === 'cover') await drawCover(g, page, k, ori, getBmp);
  else if (page.style === 'page') await drawPageFrame(g, page, k, ori, getBmp);
  else await drawMatrix(g, page, k, ori, getBmp);
  drawPill(g, k, P, `Strona ${page.no} z ${page.total}`, page.style === 'page');
}

/* ===================== 9. PDF (własny zapis, bez bibliotek) ===================== */
const pf = n => (Math.round(n * 1000) / 1000).toString();
// items: [{ strips:[{jpeg:Blob, iw, ih, y0, y1}], fullH, pw, ph, x, y, w, h }] – wymiary w punktach (pt), x/y od lewego górnego rogu.
// Strona może składać się z kilku pasków-obrazów ułożonych jeden pod drugim (dzięki temu nie ma limitu wielkości płótna).
function buildPdf(items) {
  const enc = new TextEncoder(), parts = [], offs = [];
  let pos = 0;
  const push = x => { const b = typeof x === 'string' ? enc.encode(x) : x; parts.push(b); pos += (b.size !== undefined ? b.size : b.length); };
  push(new Uint8Array([0x25, 0x50, 0x44, 0x46, 0x2D, 0x31, 0x2E, 0x34, 0x0A, 0x25, 0xE2, 0xE3, 0xCF, 0xD3, 0x0A]));
  let next = 3;
  items.forEach(it => { it.no = next++; it.ct = next++; it.im = it.strips.map(() => next++); });
  const obj = (n, body) => { offs[n] = pos; push(`${n} 0 obj\n${body}\nendobj\n`); };
  obj(1, '<< /Type /Catalog /Pages 2 0 R >>');
  obj(2, `<< /Type /Pages /Kids [${items.map(it => `${it.no} 0 R`).join(' ')}] /Count ${items.length} >>`);
  items.forEach(it => {
    obj(it.no, `<< /Type /Page /Parent 2 0 R /MediaBox [0 0 ${pf(it.pw)} ${pf(it.ph)}] /Resources << /XObject << ${it.strips.map((_, j) => `/Im${j} ${it.im[j]} 0 R`).join(' ')} >> >> /Contents ${it.ct} 0 R >>`);
    let cs = '';
    it.strips.forEach((st, j) => {                        // pasek zajmuje wiersze [y0, y1) rastra; pozycje liczone z jednej skali, bez kumulacji błędu
      const top = it.y + st.y0 / it.fullH * it.h, bot = it.y + st.y1 / it.fullH * it.h;
      cs += `q ${pf(it.w)} 0 0 ${pf(bot - top)} ${pf(it.x)} ${pf(it.ph - bot)} cm /Im${j} Do Q\n`;
    });
    if (it.marks) cs += it.marks;                        // krzyżyki cięcia rysujemy wektorowo, nad obrazem
    obj(it.ct, `<< /Length ${cs.length} >>\nstream\n${cs}endstream`);
    it.strips.forEach((st, j) => {
      offs[it.im[j]] = pos;
      push(`${it.im[j]} 0 obj\n<< /Type /XObject /Subtype /Image /Width ${st.iw} /Height ${st.ih} /ColorSpace /DeviceRGB /BitsPerComponent 8 /Filter /DCTDecode /Length ${st.jpeg.size} >>\nstream\n`);
      push(st.jpeg); push('\nendstream\nendobj\n');
    });
  });
  const total = next, xref = pos;
  let x = `xref\n0 ${total}\n0000000000 65535 f \n`;
  for (let n = 1; n < total; n++) x += String(offs[n]).padStart(10, '0') + ' 00000 n \n';
  push(x + `trailer\n<< /Size ${total} /Root 1 0 R >>\nstartxref\n${xref}\n%%EOF\n`);
  return new Blob(parts, { type: 'application/pdf' });   // Blob z części: przeglądarka trzyma dane na dysku, nie w jednej wielkiej tablicy
}

/* ---------- eksport w maksymalnej jakości bez zawieszania ---------- */
const PDF_MAX_DPI = 600;                  // sufit trybu „Maksymalna” (tyle wystarcza do druku fotograficznego)
const STRIP_AREA = 8e6;                   // piksele w jednym pasku strony (~32 MB pamięci na pasek)
const STRIP_OVERLAP_PT = 1.5;             // paski zachodzą na siebie o 1,5 pt (liczone w pikselach wg DPI) – przeglądarki PDF nie pokazują szwu
const JPEG_Q = 0.95;
const BYTES_PER_PX = 0.22;                // zachowawcze oszacowanie rozmiaru JPEG do budżetu pliku
const sizeBudget = () => clamp((navigator.deviceMemory || 4) * 150e6, 300e6, 1200e6);
let canvasCaps = null;
// Największa szerokość płótna, jaką przeglądarka faktycznie obsłuży (cienkie płótna – test nic nie kosztuje).
function probeCanvasWidth() {
  if (canvasCaps) return canvasCaps;
  let maxW = 4096;
  for (const w of [32767, 16384, 8192, 4096]) {
    try {
      const c = document.createElement('canvas'); c.width = w; c.height = 2;
      const g = c.getContext('2d'); if (!g) continue;
      g.fillStyle = '#ff0000'; g.fillRect(w - 2, 0, 2, 2);
      const ok = g.getImageData(w - 1, 1, 1, 1).data[0] === 255;
      c.width = c.height = 1;
      if (ok) { maxW = w; break; }
    } catch (e) { /* za szerokie */ }
  }
  return (canvasCaps = { maxW });
}
function stripPlan(pxW, pxH) {
  const area = window.__STRIP_AREA || STRIP_AREA;        // window.__STRIP_AREA: haczyk testowy
  const maxH = Math.max(256, Math.floor(area / pxW)), n = Math.max(1, Math.ceil(pxH / maxH));
  return { n, each: Math.ceil(pxH / n) };
}
async function renderStripJpeg(page, pxW, y0, h, getBmp, bleedFrac) {
  const canvas = document.createElement('canvas');
  try {
    await renderPage(page, canvas, pxW, getBmp, { y0, h }, bleedFrac);
    return await toBlob(canvas, 'image/jpeg', JPEG_Q);
  } catch (e) {
    if (e.name === 'AbortError') throw e;
    const oom = new Error(`nie udało się wyrenderować paska strony (${e.message})`); oom.oom = true; throw oom;
  } finally { canvas.width = canvas.height = 1; }       // zwalnia pamięć od razu, bez czekania na GC
}
let exportCancelled = false, lastExport = null;
// Krzyżyki cięcia: cztery narożniki, rysowane poza obszarem spadu, w punktach PDF.
function cropMarkStream(pw, ph, x, y, w, h, bleedPt) {
  const len = 14, gap = Math.max(2, bleedPt), lw = 0.4;
  const xs = [x, x + w], ys = [y, y + h];                 // y liczone od dołu strony
  let cs = `q 0 0 0 RG ${lw} w\n`;
  for (const cx of xs) for (const cy of ys) {
    const dx = cx === x ? -1 : 1, dy = cy === y ? -1 : 1;
    cs += `${pf(cx + dx * gap)} ${pf(cy)} m ${pf(cx + dx * (gap + len))} ${pf(cy)} l S\n`;
    cs += `${pf(cx)} ${pf(cy + dy * gap)} m ${pf(cx)} ${pf(cy + dy * (gap + len))} l S\n`;
  }
  return cs + 'Q\n';
}
async function buildAlbumPdf(S, dpi, mmW, mmH) {
  const ori = layoutState.orientation, P = PAGE[ori];
  const bleed = Math.max(0, Math.min(5, S.bleedMm || 0)), markMm = S.cropMarks ? 8 : 0;
  // Spad: raster robimy większy o spad z każdej strony, w tej samej rozdzielczości na milimetr.
  const mmFullW = mmW + 2 * bleed, mmFullH = mmH + 2 * bleed;
  const pxW = Math.round(mmFullW / 25.4 * dpi), pxH = Math.round(pxW * (mmFullH / mmFullW));
  const { n: nStr, each } = stripPlan(pxW, pxH), totalStrips = pages.length * nStr, overlap = Math.max(2, Math.ceil(dpi * STRIP_OVERLAP_PT / 72));
  const mm2pt = 72 / 25.4;
  const items = [], pw = (mmFullW + 2 * markMm) * mm2pt, ph = (mmFullH + 2 * markMm) * mm2pt, punch = 20, t0 = Date.now();
  let done = 0;
  for (let i = 0; i < pages.length; i++) {
    const cache = new Map();                              // obrazy strony dekodujemy raz i dzielimy między paski
    const getBmp = async (key, blob, maxW) => { const ck = key + '@' + (maxW || 0); if (!cache.has(ck)) cache.set(ck, await decodeScaled(blob, maxW)); return cache.get(ck); };
    const strips = [];
    try {
      for (let j = 0; j < nStr; j++) {
        if (exportCancelled) throw abortErr();
        const y0 = j * each, y1 = Math.min(pxH, (j + 1) * each), ys = Math.max(0, y0 - overlap);
        const eta = done ? ` · zostało ok. ${fmtDur((Date.now() - t0) / done * (totalStrips - done))}` : '';
        setLoader(`Strona ${i + 1} z ${pages.length}${nStr > 1 ? ` · pasek ${j + 1}/${nStr}` : ''}\n${dpi} DPI · ${pxW} × ${pxH} px${eta}`, done / totalStrips * 100);
        strips.push({ jpeg: await renderStripJpeg(pages[i], pxW, ys, y1 - ys, getBmp, bleed / mmFullW), iw: pxW, ih: y1 - ys, y0: ys, y1 });
        done++;
        await yieldUi();                                  // ekran i przycisk „Anuluj” pozostają responsywne
      }
    } finally { cache.forEach(b => b.close && b.close()); cache.clear(); }
    const bw = mmFullW * mm2pt, bh = mmFullH * mm2pt, m0 = markMm * mm2pt;
    const it = { strips, fullH: pxH, pw, ph, x: m0, y: m0, w: bw, h: bh };
    if (S.cropMarks) it.marks = cropMarkStream(pw, ph, m0 + bleed * mm2pt, m0 + bleed * mm2pt, mmW * mm2pt, mmH * mm2pt, bleed * mm2pt);
    if (S.pdfFormat.includes('holes') && layoutState.frameStyle !== 'page') {                  // miejsce na dziurki: na przemian z lewej/prawej (albo z góry/dołu)
      const m = punch * mm2pt, odd = (i + 1) % 2 === 1;
      const availW = (S.pdfFormat === 'a4_holes_left' ? pw - m : pw), availH = (S.pdfFormat === 'a4_holes_top' ? ph - m : ph);
      const fit = Math.min(availW / bw, availH / bh); it.w = bw * fit; it.h = bh * fit;
      const ox = (availW - it.w) / 2, oy = (availH - it.h) / 2;
      if (S.pdfFormat === 'a4_holes_left') { it.x = odd ? m + ox : ox; it.y = oy; } else { it.y = odd ? m + oy : oy; it.x = ox; }
      it.marks = null;                                    // przy dziurkach znaczniki cięcia nie mają sensu
    }
    items.push(it);
  }
  setLoader('Zapisuję plik…', 100);
  await yieldUi();
  return { blob: buildPdf(items), dpi, pxW, pxH, strips: nStr };
}
async function exportPdf() {
  if (!pages.length || running) return;
  const S = readSettings(), ori = layoutState.orientation;
  const base = S.pdfFormat.startsWith('a4') ? 'a4' : 'a2';
  let mmW = base === 'a4' ? 210 : 420, mmH = base === 'a4' ? 297 : 594;
  if (ori === 'landscape') [mmW, mmH] = [mmH, mmW];
  const inW = mmW / 25.4, inH = mmH / 25.4;
  let dpi = S.dpi || PDF_MAX_DPI;                         // „Maksymalna” = najwyższe DPI, które urządzenie zmieści
  const byWidth = Math.floor(probeCanvasWidth().maxW / inW);
  if (dpi > byWidth) { dpi = byWidth; toast(`Przeglądarka obsługuje płótna do ${probeCanvasWidth().maxW} px szerokości – ustawiono ${dpi} DPI.`, 6000); }
  if (!S.dpi) {                                           // budżet rozmiaru pliku dotyczy tylko trybu automatycznego
    const byBudget = Math.floor(Math.sqrt(sizeBudget() / (pages.length * inW * inH * BYTES_PER_PX)));
    if (dpi > byBudget) { dpi = Math.max(150, byBudget); toast(`Przy ${pages.length} stronach w tym formacie plik miałby ponad ${Math.round(sizeBudget() / 1e6)} MB – ustawiono ${dpi} DPI. Wybierz DPI ręcznie, jeśli chcesz więcej.`, 8000); }
  }
  exportCancelled = false;
  showLoader('Składam PDF…', `Format ${S.pdfFormat.toUpperCase()}, ${dpi} DPI.\nStrony powstają paskami, więc aplikacja się nie zawiesi.`, () => { exportCancelled = true; });
  await sleep(60);
  try {
    let res = null;
    for (let tryNo = 0; !res; tryNo++) {
      try { res = await buildAlbumPdf(S, dpi, mmW, mmH); }
      catch (e) {
        if (e.name === 'AbortError' || !e.oom || tryNo >= 3 || dpi <= 150) throw e;
        dpi = Math.max(150, Math.floor(dpi * 0.75));
        log(`Eksport PDF: ${e.message} – ponawiam w ${dpi} DPI`, 'warn');
        toast(`Urządzeniu zabrakło pamięci – obniżam jakość do ${dpi} DPI i zaczynam od nowa.`, 7000);
        await yieldUi();
      }
    }
    pdfBlob = res.blob; lastExport = res;
    log(`PDF gotowy: ${pages.length} stron, ${res.dpi} DPI, ${res.pxW}×${res.pxH} px na stronę, pasków na stronę: ${res.strips}`
      + (S.bleedMm ? `, spad ${S.bleedMm} mm` : '') + (S.cropMarks ? ', znaczniki cięcia' : '') + `, ${(pdfBlob.size / 1048576).toFixed(1)} MB`);
    onPdfReady();
  } catch (e) {
    if (e.name === 'AbortError') toast('Eksport PDF anulowany.');
    else { log(`Eksport PDF: ${e.message}`, 'error'); toast('Błąd eksportu PDF: ' + e.message + ' Spróbuj niższej jakości (DPI).', 7000); }
  } finally { hideLoader(); }
}

// Karty i przełączniki ([data-v]) sterują ukrytymi <select> – reszta aplikacji czyta je jak dotąd.
function bindChoice(group, select) {
  const sync = () => group.querySelectorAll('[data-v]').forEach(b => { const on = b.dataset.v === select.value; b.classList.toggle('on', on); b.setAttribute('aria-checked', String(on)); });
  group.addEventListener('click', e => {
    const b = e.target.closest('[data-v]'); if (!b || !group.contains(b)) return;
    if (select.value !== b.dataset.v) { select.value = b.dataset.v; select.dispatchEvent(new Event('change', { bubbles: true })); }
    sync();
  });
  select.addEventListener('change', sync);
  sync(); return sync;
}

const SEC_PER_IMAGE = { '4K': 26, '2K': 16, '1K': 9 };
function genPlan() {
  const S = readSettings(), nd = new Set(photos.map(p => p.dateStr)).size;
  const stories = S.doStory ? nd : 0, covers = S.doStory ? nd * Math.min(S.storyPages, 4) : 0;
  const stamps = S.stampEvery ? Math.floor(photos.length / S.stampEvery) : 0;
  const stronyZdjec = () => {                                   // ta sama zasada parowania co w układzie stron
    let n = 0;
    const byDay = new Map();
    photos.forEach(p => { if (!byDay.has(p.dateStr)) byDay.set(p.dateStr, []); byDay.get(p.dateStr).push(p); });
    for (const list of byDay.values()) {
      let i = 0;
      while (i < list.length) {
        const a = list[i], b = list[i + 1];
        let para = false;
        if (b && S.layoutMode === 'classic') para = a.orientation === 'portrait' && b.orientation === 'portrait';
        else if (b && S.layoutMode === 'smart') para = (S.orientation === 'portrait' && a.orientation === 'landscape' && b.orientation === 'landscape')
          || (S.orientation === 'landscape' && a.orientation === 'portrait' && b.orientation === 'portrait');
        i += para ? 2 : 1; n++;
      }
    }
    return n;
  };
  const strony = S.frameStyle === 'page' ? stronyZdjec() : 0;
  const gotowe = S.frameStyle === 'page'
    ? days.flatMap(d => d.matrix || []).filter(pg => pg.deco && pg.deco.kind === 'ai').length
    : photos.filter(p => p.frame && p.frame.kind === 'ai').length;
  const ramek = S.frameStyle === 'page' ? Math.max(0, strony - gotowe) : Math.max(0, photos.length - gotowe);
  const images = ramek + covers + stamps;
  const perImg = (SEC_PER_IMAGE[S.imageSize] || 14) * (/pro/.test(S.imageModel) ? 1.6 : 1);
  const sec = (images * perImg + stories * 12) / Math.max(1, Math.min(S.concurrency, 4));
  return { dni: nd, zdjecia: photos.length, gotowe, stories, covers, stamps, images, ramek, styl: S.frameStyle, zapytania: images + stories, sec };
}

// Przypisuje zdjęciu dzień: istniejący (z listy) albo zupełnie nowy (wpisany ręcznie), i wstawia je w odpowiednie miejsce.
function setPhotoDay(photo, dateStr) {
  if (!dateStr) return;
  photo.dateStr = dateStr; photo.exifError = false;
  photos.splice(photos.indexOf(photo), 1);
  let last = -1;
  for (let j = photos.length - 1; j >= 0; j--) if (photos[j].dateStr === dateStr) { last = j; break; }
  if (last !== -1) photos.splice(last + 1, 0, photo);         // dzień już istnieje: na jego koniec
  else {
    const first = photos.findIndex(x => x.dateStr > dateStr); // nowy dzień: chronologicznie między pozostałymi
    if (first === -1) photos.push(photo); else photos.splice(first, 0, photo);
  }
  renderSortUI();
}
function renderSortUI() {
  updateGenUi();
  if (!photos.length) { renderEmpty(); setStatus('Brak zdjęć w pamięci.'); $('#dropTitle').textContent = 'Przeciągnij zdjęcia tutaj albo kliknij, aby wybrać'; return; }
  dropZone.classList.add('compact');
  $('#dropTitle').textContent = 'Dopisz kolejne zdjęcia — przeciągnij je tutaj albo kliknij';
  sortHost.innerHTML = '';
  const wrap = el('div', { class: 'sort-wrap' });
  let cur = null, dayNo = 0;
  photos.forEach((p, i) => {
    if (p.dateStr !== cur) {
      cur = p.dateStr; dayNo++;
      wrap.append(el('div', { class: 'day-head' }, `Dzień ${dayNo}`, el('small', {}, fmtDate(cur))));
    }
    const item = el('div', { class: 'sort-item' + (p.exifError ? ' error' : '') });
    const input = el('input', { type: 'number', class: 'sort-input', value: i + 1, min: 1, max: photos.length, 'aria-label': 'Numer zdjęcia w albumie' });
    input.addEventListener('change', ev => {
      let nv = parseInt(ev.target.value, 10) - 1;
      if (isNaN(nv)) return renderSortUI();
      nv = clamp(nv, 0, photos.length - 1);
      const [moved] = photos.splice(i, 1); photos.splice(nv, 0, moved); renderSortUI();
    });
    const del = el('button', { class: 'sort-del', title: 'Usuń zdjęcie z albumu', 'aria-label': 'Usuń zdjęcie', onclick: () => { URL.revokeObjectURL(p.thumbUrl); photos.splice(photos.indexOf(p), 1); renderSortUI(); } }, '✕');
    item.append(el('img', { class: 'sort-img', src: p.thumbUrl, alt: p.name, loading: 'lazy' }), input, del);
    if (p.exifError) {
      const sel = el('select', { 'aria-label': 'Wybierz dzień dla zdjęcia bez daty' }, el('option', { value: '', disabled: true, selected: true }, 'Wybierz…'));
      [...new Set(photos.map(x => x.dateStr))].forEach(d => sel.append(el('option', { value: d }, fmtDate(d))));
      sel.addEventListener('change', ev => setPhotoDay(p, ev.target.value));
      // Druga droga: wpisanie dowolnej daty, gdy żaden z istniejących dni nie pasuje (albo gdy żadne zdjęcie nie ma daty).
      const dat = el('input', { type: 'date', 'aria-label': 'Wpisz własną datę zdjęcia' });
      dat.addEventListener('change', ev => {
        const v = ev.target.value;                            // YYYY-MM-DD z pola daty -> klucz dnia YYYY.MM.DD
        if (!/^\d{4}-\d{2}-\d{2}$/.test(v)) return;
        setPhotoDay(p, v.replace(/-/g, '.'));
      });
      item.append(el('div', { class: 'sort-warn' }, el('span', {}, 'Brak daty – wybierz dzień:'), sel,
        el('span', { style: 'color:var(--muted)' }, 'albo wpisz własną:'), dat));
    }
    wrap.append(item);
  });
  sortHost.append(wrap);
  setStatus(`<strong>Wgrano ${photos.length} z ${MAX_PHOTOS} ${plural(photos.length, 'zdjęcia', 'zdjęć', 'zdjęć')} · ${dayNo} ${plural(dayNo, 'dzień', 'dni', 'dni')}.</strong> Ułóż kolejność numerami albo dopisz kolejne pliki.`);
  markStale();
}

let wall = null, taskReg = new Map();
function makeTile(label, ico, thumb) {
  const img = el('img', { alt: label, src: thumb || EMPTY_PX }), tag = el('span', { class: 'tag' });
  const box = el('div', { class: 'tile wait' + (thumb ? ' has' : ''), title: label }, el('div', { class: 'ico' }, ico), img, el('span', { class: 'kind' }, ico), tag);
  const t = {
    el: box, state: 'wait',
    set(state, info) {
      info = info || {}; t.state = state;
      const has = box.classList.contains('has') || !!info.url;
      box.className = 'tile ' + state + (has ? ' has' : '');
      if (info.url) img.src = info.url;
      if (info.tag) tag.textContent = info.tag;
      if (info.msg) box.title = label + ' – ' + info.msg;
      if (info.model && wall) wall.model = (IMAGE_MODELS[info.model] || {}).name || info.model;
      updateWall();
    }
  };
  return t;
}
// Zadanie ma stały klucz: ponowienie (ręczne lub automatyczne) używa tego samego kafelka zamiast dokładać nowy.
function makeTask(key, kind, label, ico, thumb, day, exec) {
  let tile; const old = taskReg.get(key);
  if (old) { tile = old.tile; tile.set('wait'); }
  else { tile = makeTile(label, ico, thumb); if (day && day._grid) day._grid.append(tile.el); }
  const t = { key, kind, label, tile, exec: () => exec(tile) };
  taskReg.set(key, t);
  return t;
}
function updateWall() {
  if (!wall || !$('#wBar')) return;
  const T = wall.tasks, isFin = t => t.tile.state === 'ok' || t.tile.state === 'err';
  const fin = T.filter(isFin).length, errs = T.filter(t => t.tile.state === 'err').length;
  $('#wBar').style.width = (T.length ? fin / T.length * 100 : 0) + '%';
  const kinds = { story: '📖 opowieści', cover: '🖼️ ilustracje', frame: '🍌 ramki', stamp: '🕒 znaczniki' };
  $('#wStats').innerHTML = Object.entries(kinds).map(([k, lbl]) => {
    const a = T.filter(t => t.kind === k);
    return a.length ? `<span class="stat">${lbl} ${a.filter(isFin).length}/${a.length}</span>` : '';
  }).join('') + (errs ? `<span class="stat err">błędy: ${errs}</span>` : '');
  const eta = fin ? (Date.now() - wall.t0) / fin * (T.length - fin) : null;
  $('#wSub').textContent = `${wall.label} · ${fin}/${T.length}${eta != null && fin < T.length ? ' · zostało ok. ' + fmtDur(eta) : ''}${wall.model ? ' · ostatnio: ' + wall.model : ''}`;
}

async function pool(items, n, signal, worker) {
  n = Math.max(1, Math.min(n, concCap));                  // concCap spada do 1 po limicie chwilowym z API
  let i = 0;
  await Promise.all(Array.from({ length: Math.max(1, n) }, async () => {
    while (i < items.length && !signal.aborted) { const it = items[i++]; await worker(it); }
  }));
}
async function runStage(tasks, label, S) {
  wall = { tasks, t0: Date.now(), label, model: '' };
  updateWall();
  await pool(tasks, S.concurrency, runCtl.signal, async t => {
    try { await t.exec(); }
    catch (e) {
      if (e.name === 'AbortError') { t.tile.set('wait'); return; }
      log(`${t.label}: ${e.message}`, 'error');
      if (t.tile.state !== 'err') t.tile.set('err', { msg: e.message });
      if (e.fatal) { fatalMsg = e.message; runCtl.abort(); }
    }
  });
}


// Budowniczowie zadań: te same funkcje służą pierwszemu przebiegowi i automatycznym rundom naprawczym.
const T = {
  story: (d, S, sig) => makeTask('story:' + d.index, 'story', `Dzień ${d.index} – opowieść`, '📖', null, d, tile => storyTask(d, S, sig, tile)),
  cover: (d, ci, S, sig) => makeTask(`cover:${d.index}:${ci}`, 'cover', `Dzień ${d.index} – ilustracja ${ci + 1}`, '🖼️', null, d, tile => coverTask(d, d.covers[ci], S, sig, tile)),
  frame: (p, S, sig) => makeTask('frame:' + p.id, 'frame', `Ramka ${p.n}`, '🍌', p.thumbUrl, days.find(d => d.index === p.dayIdx), tile => frameTask(p, S, sig, tile)),
  pageFrame: (pg, S, sig) => makeTask('deco:' + pg.id, 'frame', `Strona ${pg.tag}`, '🖼️', pg.photos[0].thumbUrl, pg.day, tile => pageFrameTask(pg, S, sig, tile)),
  stamp: (p, S, sig) => makeTask('stamp:' + p.id, 'stamp', `Znacznik ${p.timeStr}`, '🕒', null, days.find(d => d.index === p.dayIdx), tile => stampTask(p, days.find(d => d.index === p.dayIdx), S, sig, tile))
};
const AUTO_ROUNDS = 3;
// Automatyczne rundy naprawcze: wszystko, co nie ma wyniku z AI, jest ponawiane bez klikania (z przerwą na limity API,
// a w kolejnych rundach w niższej rozdzielczości), aż każde zdjęcie dostanie ramkę albo skończą się rundy.
async function autoRepair(S) {
  const sig = runCtl.signal;
  for (let round = 1; round <= AUTO_ROUNDS; round++) {
    if (sig.aborted || fatalMsg || !S.autoRepair) return;
    const badStories = S.doStory ? days.filter(d => d.storyFailed) : [];
    const badFrames = S.frameStyle === 'page'
      ? days.flatMap(d => d.matrix).filter(pg => !pg.deco || pg.deco.kind !== 'ai')
      : photos.filter(p => !p.frame || p.frame.kind !== 'ai');
    const badCovers = []; if (S.doStory) days.forEach(d => { if (!d.storyFailed) d.covers.forEach((cv, ci) => { if (!cv.blob) badCovers.push([d, ci]); }); });
    const badStamps = photos.filter(p => wantsStamp(p, S) && !(p.stamp && p.stamp.blob));
    const total = badStories.length + badFrames.length + badCovers.length + badStamps.length;
    if (!total) return;
    log(`Naprawa automatyczna ${round}/${AUTO_ROUNDS}: brakuje ${total} elementów (opowieści ${badStories.length}, ramki ${badFrames.length}, ilustracje ${badCovers.length}, znaczniki ${badStamps.length})`, 'warn');
    const until = Date.now() + Math.max(cooldownUntil - Date.now(), 0) + 8000 * round;      // przerwa: API odetchnie po limitach
    while (Date.now() < until && !sig.aborted) {
      const sub = $('#wSub'); if (sub) sub.textContent = `Naprawa automatyczna ${round}/${AUTO_ROUNDS}: ponawiam ${total} ${plural(total, 'element', 'elementy', 'elementów')} za ${Math.ceil((until - Date.now()) / 1000)} s…`;
      try { await sleepAbortable(1000, sig); } catch (e) { return; }
    }
    const S2 = round >= 2 ? Object.assign({}, S, { imageSize: downSize(S.imageSize) }) : S;   // od drugiej rundy: niższa rozdzielczość = większa szansa
    if (badStories.length) await runStage(badStories.map(d => T.story(d, S, sig)), `Naprawa ${round}/${AUTO_ROUNDS} · opowieści`, S);
    if (sig.aborted || fatalMsg) return;
    const tasks = [];
    if (S.doStory) days.forEach(d => { if (!d.storyFailed) d.covers.forEach((cv, ci) => { if (!cv.blob) tasks.push(T.cover(d, ci, S2, sig)); }); });
    if (S.frameStyle === 'page') days.flatMap(d => d.matrix).filter(pg => !pg.deco || pg.deco.kind !== 'ai').forEach(pg => tasks.push(T.pageFrame(pg, S2, sig)));
    else photos.filter(p => !p.frame || p.frame.kind !== 'ai').forEach(p => tasks.push(T.frame(p, S2, sig)));
    photos.filter(p => wantsStamp(p, S) && !(p.stamp && p.stamp.blob)).forEach(p => tasks.push(T.stamp(p, S2, sig)));
    if (tasks.length) await runStage(tasks, `Naprawa ${round}/${AUTO_ROUNDS} · ilustracje i ramki`, S2);
  }
}

// Trzy próbne ramki: styl i jakość widać przed puszczeniem całego albumu.
async function runSample() {
  if (running || importing || !photos.length) return;
  const S = readSettings(); fatalMsg = '';
  planLayout(S);
  const pageStyle = S.frameStyle === 'page';
  const all = pageStyle ? days.flatMap(d => d.matrix) : photos;
  const step = Math.max(1, Math.floor(all.length / 3));
  const list = [0, step, step * 2].map(i => all[Math.min(i, all.length - 1)]).filter((x, i, a) => a.indexOf(x) === i);
  running = true; runCtl = new AbortController(); updateGenUi();
  const out = $('#sampleOut');
  out.innerHTML = '';
  out.append(el('b', { style: 'font:700 14px var(--sans)' }, `Próbka: ${list.length} ${plural(list.length, 'ramka', 'ramki', 'ramek')}`));
  const row = el('div', { class: 'row' }); out.append(row);
  const imgs = list.map(() => { const i = el('img', { alt: 'Próbna ramka' }); row.append(i); return i; });
  showLoader('Rysuję próbkę…', `${list.length} ${plural(list.length, 'ramka', 'ramki', 'ramek')} w ustawieniach z kroku 05.`, () => runCtl.abort());
  let done = 0;
  await pool(list, S.concurrency, runCtl.signal, async (p, i) => {
    try {
      if (pageStyle) { await pageFrameTask(p, S, runCtl.signal, NO_TILE); imgs[i].src = p.deco.thumbUrl || p.deco.url; }
      else { await frameTask(p, S, runCtl.signal, NO_TILE); imgs[i].src = p.frame.thumbUrl || p.frame.url; }
    }
    catch (e) { if (e.name !== 'AbortError') { fatalMsg = e.message; if (e.fatal) runCtl.abort(); } }
    setLoader(`${++done}/${list.length}`, done / list.length * 100);
  });
  running = false; hideLoader(); updateGenUi();
  const ai = list.filter(x => (pageStyle ? x.deco : x.frame) && (pageStyle ? x.deco : x.frame).kind === 'ai').length;
  out.append(el('p', { class: 'hint' }, ai === list.length
    ? 'Podoba się? Kliknij „Generuj fotoksiążkę” — te ramki zostaną użyte, nie policzę ich drugi raz.'
    : `Udało się ${ai} z ${list.length}. ${fatalMsg || 'Sprawdź ustawienia albo spróbuj ponownie za chwilę.'}`));
  if (fatalMsg) toast(fatalMsg, 9000);
  out.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
}

async function startGeneration() {
  if (running || importing || !photos.length) return;
  const S = readSettings(); fatalMsg = '';
  planLayout(S);
  running = true; runCtl = new AbortController(); pdfBlob = null; updateGenUi();
  $('#sampleOut').innerHTML = '';
  $('#downloadBtn').style.display = 'none';
  buildWall();
  workspace.scrollIntoView({ behavior: 'smooth', block: 'start' });
  log(`Start: ${photos.length} ${plural(photos.length, 'zdjęcie', 'zdjęcia', 'zdjęć')}, ${days.length} ${plural(days.length, 'dzień', 'dni', 'dni')}, model ${S.imageModel}, ${S.imageSize || '-'}, tryb ${S.frameMode}`);
  const sig = runCtl.signal;
  try {
    if (S.doStory) await runStage(days.map(d => T.story(d, S, sig)), 'Etap 1/2 · opowieści dni', S);
    if (!sig.aborted) {
      const t2 = [];
      days.forEach(d => {
        if (S.doStory && !d.storyFailed) d.covers.forEach((cv, ci) => t2.push(T.cover(d, ci, S, sig)));
        if (S.frameStyle === 'page') d.matrix.forEach(pg => { if (!(pg.deco && pg.deco.kind === 'ai')) t2.push(T.pageFrame(pg, S, sig)); });
        else d.photos.forEach(p => { if (!(p.frame && p.frame.kind === 'ai')) t2.push(T.frame(p, S, sig)); });   // ramki z próbki zostają
        d.photos.forEach(p => { if (wantsStamp(p, S)) t2.push(T.stamp(p, S, sig)); });
      });
      await runStage(t2, S.doStory ? 'Etap 2/2 · ilustracje i ramki' : 'Ramki i znaczniki', S);
    }
    await autoRepair(S);
  } catch (e) { log('Generowanie: ' + e.message, 'error'); }
  const stopped = runCtl.signal.aborted;
  running = false; wall = null;
  if (fatalMsg) toast(fatalMsg, 10000);
  else if (stopped) toast('Zatrzymano. Brakujące ramki zastąpiono lokalnymi – możesz je ponowić w podglądzie.', 7000);
  showLoader('Składanie stron…', 'Przygotowuję podgląd albumu.');
  await sleep(30);
  try {
    if (S.frameStyle === 'page') { for (const d of days) for (const pg of d.matrix) if (!pg.deco) await makeLocalPageDeco(pg, S); }
    else for (const p of photos) if (!p.frame) await makeLocalFrame(p, S);
    days.forEach(d => { if (S.doStory && !d.covers.length) setCovers(d, [STORY_FALLBACK]); });
    buildPages(S);
  } finally { hideLoader(); }
  renderPagesView(true);
  updateGenUi();
}

/* ---------- podgląd stron ---------- */
let pagesObserver = null, headEl = null;
function frameBtnContent(p) {
  const badge = p.frame && p.frame.kind === 'ai'
    ? el('span', { class: 'badge' }, (IMAGE_MODELS[p.frame.model] || {}).short || 'AI')
    : el('span', { class: 'badge local' }, 'lokalna');
  return [`Ramka ${p.n}`, badge];
}
function updateFrameBtn(p) { if (p._btn) { p._btn.disabled = false; p._btn.replaceChildren(...frameBtnContent(p)); } }
function pagesHeader() {
  const pageStyle = layoutState.frameStyle === 'page';
  const units = pageStyle ? pages.filter(pg => pg.type === 'matrix') : photos;
  const local = pageStyle ? units.filter(pg => !pg.deco || pg.deco.kind === 'local').length : photos.filter(p => !p.frame || p.frame.kind === 'local').length;
  const ai = units.length - local;
  return el('div', { class: 'panel sticky-head' },
    el('div', {},
      el('div', { class: 'panel-title' }, `Gotowe: ${pages.length} ${plural(pages.length, 'strona', 'strony', 'stron')} · ramki AI ${ai}/${units.length}`),
      el('div', { class: 'hint', style: 'margin-top:4px' }, local ? `${local} ${plural(local, 'ramka jest lokalna', 'ramki są lokalne', 'ramek jest lokalnych')} (bez AI). ${plural(local, 'Możesz ją', 'Możesz je', 'Możesz je')} ponowić jednym kliknięciem.` : 'Przejrzyj strony, a potem wygeneruj PDF.')),
    el('div', { style: 'display:flex;gap:8px;flex-wrap:wrap' },
      local ? el('button', { class: 'btn btn-soft btn-sm', onclick: retryLocal }, `Ponów nieudane (${local})`) : null,
      el('button', { class: 'btn btn-ok btn-sm', onclick: exportPdf }, 'Generuj PDF'),
      pdfBlob ? el('button', { class: 'btn btn-primary btn-sm', onclick: downloadPdf }, `Pobierz PDF (${(pdfBlob.size / 1048576).toFixed(1)} MB${lastExport ? ', ' + lastExport.dpi + ' DPI' : ''})`) : null));
}
function refreshHeader() { if (headEl && headEl.isConnected) { const n = pagesHeader(); headEl.replaceWith(n); headEl = n; } }
function pageCard(pg) {
  const land = layoutState.orientation === 'landscape';
  const img = el('img', { alt: `Strona ${pg.no}` });
  const wrap = el('div', { class: 'page-wrapper' + (land ? ' land' : '') }, img, el('div', { class: 'ph' }, `Strona ${pg.no}…`));
  const tools = el('div', { class: 'page-tools' });
  if (pg.type === 'matrix' && pg.style === 'page') tools.append(el('button', { class: 'btn btn-soft btn-sm', onclick: () => regenPageFrame(pg), title: 'Wygeneruj dekoracje tej strony ponownie' },
    'Ponów ramkę strony', pg.deco && pg.deco.kind === 'ai' ? el('span', { class: 'badge' }, (IMAGE_MODELS[pg.deco.model] || {}).short || 'AI') : el('span', { class: 'badge local' }, 'lokalna')));
  else if (pg.type === 'matrix') pg.photos.forEach(p => { p._btn = el('button', { class: 'btn btn-soft btn-sm', onclick: () => regenFrame(p), title: 'Wygeneruj tę ramkę ponownie (nowa postać i poza)' }, ...frameBtnContent(p)); tools.append(p._btn); });
  else tools.append(el('button', { class: 'btn btn-soft btn-sm', onclick: () => regenDay(pg.day) }, 'Ponów opowieść dnia'));
  pg._img = img; pg._done = false; wrap._page = pg;
  pagesObserver.observe(wrap);
  return el('div', { class: 'page-card' }, wrap, tools);
}
function renderPagesView(scrollTo) {
  if (pagesObserver) pagesObserver.disconnect();
  $$('.page-wrapper img', workspace).forEach(i => { if (i.dataset.url) URL.revokeObjectURL(i.dataset.url); });
  pagesObserver = new IntersectionObserver(entries => entries.forEach(en => {
    const pg = en.target._page;
    if (en.isIntersecting && pg && !pg._done) { pg._done = true; renderPreview(pg, pg._img); }
  }), { rootMargin: '1000px 0px' });
  workspace.innerHTML = '';
  headEl = pagesHeader(); workspace.append(headEl);
  pages.forEach(pg => workspace.append(pageCard(pg)));
  workspace.append(el('button', { class: 'btn btn-ok btn-big', style: 'margin:20px 0 10px', onclick: exportPdf }, 'Generuj PDF do druku'));
  if (scrollTo) workspace.scrollIntoView({ behavior: 'smooth', block: 'start' });
}
function refreshPagesFor(photo) {
  pages.filter(pg => pg.type === 'matrix' && pg.photos.includes(photo)).forEach(pg => { pg._done = true; renderPreview(pg, pg._img); });
  updateFrameBtn(photo); refreshHeader();
}

/* ---------- ponawianie pojedynczych elementów ---------- */
async function regenFrame(photo) {
  if (running) return toast('Trwa inne generowanie – poczekaj chwilę.');
  const S = readSettings();
  running = true; runCtl = new AbortController(); fatalMsg = '';
  replanPhoto(photo, S);
  if (photo._btn) { photo._btn.disabled = true; photo._btn.replaceChildren('Ramka ' + photo.n + '…'); }
  try { await frameTask(photo, S, runCtl.signal, NO_TILE); }
  catch (e) { if (e.name !== 'AbortError') toast(e.message, 8000); }
  finally { running = false; }
  refreshPagesFor(photo);
}
async function regenPageFrame(page) {
  if (running) return toast('Trwa inne generowanie – poczekaj chwilę.');
  const S = readSettings(); fatalMsg = '';
  running = true; runCtl = new AbortController();
  showLoader(`Ponawiam ramkę strony ${page.no}…`, 'Rysuję dekorację całej kartki.', () => runCtl.abort());
  page.plan = { theme: page.day.theme, subject: page.photos[0].plan.subject, pose: page.photos[0].plan.pose };
  try { await pageFrameTask(page, Object.assign({}, S, { useCache: false }), runCtl.signal, NO_TILE); }
  catch (e) { if (e.name !== 'AbortError') toast(e.message, 8000); }
  running = false; hideLoader();
  page._done = true; renderPreview(page, page._img);
  refreshHeader();
}
async function retryLocal() {
  if (running) return;
  const S = readSettings(); fatalMsg = '';
  const pageStyle = layoutState.frameStyle === 'page';
  const list = pageStyle ? pages.filter(pg => pg.type === 'matrix' && (!pg.deco || pg.deco.kind === 'local'))
    : photos.filter(p => !p.frame || p.frame.kind === 'local');
  if (!list.length) return;
  running = true; runCtl = new AbortController();
  showLoader('Ponawiam ramki…', `0/${list.length}`, () => runCtl.abort());
  let done = 0;
  await pool(list, S.concurrency, runCtl.signal, async p => {
    try { if (pageStyle) await pageFrameTask(p, Object.assign({}, S, { useCache: false }), runCtl.signal, NO_TILE); else await frameTask(p, S, runCtl.signal, NO_TILE); }
    catch (e) { if (e.name !== 'AbortError') { fatalMsg = e.message; if (e.fatal) runCtl.abort(); } }
    done++; setLoader(`${done}/${list.length}`, done / list.length * 100);
  });
  running = false; hideLoader();
  if (fatalMsg) toast(fatalMsg, 9000);
  if (pageStyle) list.forEach(pg => { pg._done = true; renderPreview(pg, pg._img); });
  else list.forEach(p => { pages.filter(pg => pg.type === 'matrix' && pg.photos.includes(p)).forEach(pg => { pg._done = true; renderPreview(pg, pg._img); }); updateFrameBtn(p); });
  refreshHeader();
}
async function regenDay(day) {
  if (running) return toast('Trwa inne generowanie – poczekaj chwilę.');
  const S = readSettings();
  running = true; runCtl = new AbortController(); fatalMsg = '';
  showLoader(`Ponawiam opowieść dnia ${day.index}…`, 'Pisanie tekstu i rysowanie ilustracji.', () => runCtl.abort());
  try {
    await storyTask(day, S, runCtl.signal, NO_TILE);
    setLoader('Rysuję ilustracje…', 40);
    if (!day.storyFailed) await pool(day.covers, S.concurrency, runCtl.signal, async cv => {
      try { await coverTask(day, cv, S, runCtl.signal, NO_TILE); } catch (e) { if (e.name !== 'AbortError') log(`Ilustracja dnia ${day.index}: ${e.message}`, 'error'); }
    });
  } catch (e) { if (e.name !== 'AbortError') toast(e.message, 8000); }
  running = false; hideLoader();
  buildPages(S);                                   // liczba stron opowieści mogła się zmienić
  const y = window.scrollY; renderPagesView(); window.scrollTo(0, y);
}

}
